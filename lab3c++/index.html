Задача 1: Асинхронное копирование 1 ГБ файла (OVERLAPPED + hEvent)
task1_async_copy_overlapped.cpp
task1_async_copy_overlapped.cpp 
#include <windows.h>
#include <iostream>
#include <iomanip>

const DWORD CHUNK_SIZE = 64 * 1024; // 64 КБ на один блок передачи

bool CreateDummyFile(const wchar_t* path, DWORD sizeMB) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;
    LARGE_INTEGER liSize;
    liSize.QuadPart = (LONGLONG)sizeMB * 1024 * 1024;
    SetFilePointerEx(hFile, liSize, NULL, FILE_BEGIN);
    SetEndOfFile(hFile);
    CloseHandle(hFile);
    return true;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    const wchar_t* srcPath = L"source_large.dat";
    const wchar_t* dstPath = L"destination_copy.dat";

    CreateDummyFile(srcPath, 1024); // Создание 1 ГБ тестового файла

    HANDLE hSrc = CreateFileW(srcPath, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_FLAG_OVERLAPPED, NULL);
    HANDLE hDst = CreateFileW(dstPath, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_FLAG_OVERLAPPED, NULL);

    LARGE_INTEGER fileSize;
    GetFileSizeEx(hSrc, &fileSize);

    HANDLE hReadEvent = CreateEventW(NULL, TRUE, FALSE, NULL);
    HANDLE hWriteEvent = CreateEventW(NULL, TRUE, FALSE, NULL);

    char* buffer = new char[CHUNK_SIZE];
    OVERLAPPED ovRead = {0};
    OVERLAPPED ovWrite = {0};
    ovRead.hEvent = hReadEvent;
    ovWrite.hEvent = hWriteEvent;

    ULONGLONG currentOffset = 0;
    DWORD bytesTransferred = 0;

    while (currentOffset < (ULONGLONG)fileSize.QuadPart) {
        DWORD bytesToRead = (DWORD)min((ULONGLONG)CHUNK_SIZE, fileSize.QuadPart - currentOffset);

        ovRead.Offset = (DWORD)(currentOffset & 0xFFFFFFFF);
        ovRead.OffsetHigh = (DWORD)(currentOffset >> 32);
        ResetEvent(hReadEvent);

        ReadFile(hSrc, buffer, bytesToRead, NULL, &ovRead);
        WaitForSingleObject(hReadEvent, INFINITE);
        GetOverlappedResult(hSrc, &ovRead, &bytesTransferred, FALSE);

        ovWrite.Offset = ovRead.Offset;
        ovWrite.OffsetHigh = ovRead.OffsetHigh;
        ResetEvent(hWriteEvent);

        WriteFile(hDst, buffer, bytesTransferred, NULL, &ovWrite);
        WaitForSingleObject(hWriteEvent, INFINITE);
        GetOverlappedResult(hDst, &ovWrite, &bytesTransferred, FALSE);

        currentOffset += bytesTransferred;
    }

    CloseHandle(hReadEvent);
    CloseHandle(hWriteEvent);
    CloseHandle(hSrc);
    CloseHandle(hDst);
    delete[] buffer;
    return 0;
}
Подробный построчный разбор команд
FILE_FLAG_OVERLAPPED Системный флаг в CreateFileW. Указывает ядру, что операции I/O должны выполняться асинхронно контроллером диска в фоновом режиме через DMA, не усыпляя вызывающий поток.
ovRead.Offset = (DWORD)(currentOffset & 0xFFFFFFFF); В асинхронном режиме скрытый файловый указатель не обновляется. Смещение обязано передаваться вручную для каждого 64 КБ блока через поля Offset и OffsetHigh.
WaitForSingleObject(hReadEvent, INFINITE); Поток уступает управление планировщику Windows до момента, пока диск не сгенерирует аппаратное прерывание и драйвер не переведет событие hEvent в сигнальное состояние.
GetOverlappedResult(hSrc, &ovRead, &bytesTransferred, FALSE); Получает точное число переданных байт от ядра и проверяет статус завершения фонового пакета ввода-вывода.
Задача 2: Чтение MFT (Резидентные vs Нерезидентные атрибуты)
task2_mft_resident_theory.cpp
Правило MFT NTFS: Запись MFT имеет фиксированный размер 1024 байта (1 КБ). Если файл меньше ~700 байт (например 200 байт), его данные хранятся прямо внутри записи MFT (резидентный). Если файл больше (2000 байт или 500 КБ), он выносится в кластеры на диске (нерезидентный), а MFT хранит цепочки адресов (Data Runs / Extents).
Разбор системного вызова FSCTL_GET_RETRIEVAL_POINTERS

1. DeviceIoControl(hFile, FSCTL_GET_RETRIEVAL_POINTERS, ...) запрашивает карту физических кластеров файла на диске.

2. Для резидентного файла вызов возвращает ошибку ERROR_HANDLE_EOF (38) — файл не имеет отдельных кластеров на томе, так как лежит в MFT.

3. Для нерезидентного файла возвращается структура RETRIEVAL_POINTERS_BUFFER со списком номеров логических кластеров (LCN).

Задача 3: Реальный физический размер файлов вне таблицы MFT
task3_mft_physical_size.cpp
task3_mft_physical_size.cpp 
#include <windows.h>
#include <iostream>

void investigateFile(const wchar_t* fileName, DWORD bytesToWrite) {
    HANDLE hFile = CreateFileW(fileName, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    char* buffer = new char[bytesToWrite];
    memset(buffer, 'A', bytesToWrite);
    DWORD bytesWritten = 0;
    WriteFile(hFile, buffer, bytesToWrite, &bytesWritten, NULL);
    CloseHandle(hFile);
    delete[] buffer;

    DWORD highSize = 0;
    DWORD physicalSize = GetCompressedFileSizeW(fileName, &highSize);

    std::wcout << L"Файл: " << fileName << std::endl;
    std::wcout << L"  Записано данных: " << bytesToWrite << L" байт." << std::endl;
    std::wcout << L"  Занято на диске: " << physicalSize << L" байт." << std::endl;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    investigateFile(L"small_file.txt", 200);   // Кейс 1: 200 байт
    investigateFile(L"large_file.txt", 2000); // Кейс 2: 2000 байт
    return 0;
}
Построчный разбор результатов
GetCompressedFileSizeW(fileName, &highSize); Опрашивает драйвер файловой системы о количестве физических байт, выделенных на диске под файл.
Кейс 1 (200 байт): Занято на диске = 0 байт! Файл целиком размещен в 1024-байтной записи MFT. Вне таблицы MFT файловая система не выделила ни одного сектора.
Кейс 2 (2000 байт): Занято на диске = 4096 байт! 2000 байт не поместились в свободное место записи MFT. Файл стал нерезидентным, и NTFS выделила 1 физический кластер (4 КБ = 4096 байт).
Задача 4: Эффект блокировки потока при синхронном I/O
task4_sync_blocking_demo.cpp
Физика блокировки: При вызове синхронного ReadFile без флага FILE_FLAG_OVERLAPPED планировщик ядра Windows переводит поток из состояния Running в состояние Waiting (Blocked). Задача 3 (Самостоятельно 2): Проецирование со смещением (Allocation Granularity)
task3_memory_mapped_offset.cpp
task3_memory_mapped_offset.cpp 
#include <windows.h>
#include <iostream>

int main() {
    SYSTEM_INFO sysInfo;
    GetSystemInfo(&sysInfo);
    DWORD allocGranularity = sysInfo.dwAllocationGranularity; // 64 КБ (65536)

    HANDLE hFile = CreateFileW(L"mapped_test.txt", GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    HANDLE hMapping = CreateFileMappingW(hFile, NULL, PAGE_READONLY, 0, 0, NULL);

    // Смещение ОБЯЗАНО быть кратно allocGranularity (64 КБ)!
    size_t targetOffset = 2 * allocGranularity; // 128 КБ
    DWORD offsetLow = (DWORD)(targetOffset & 0xFFFFFFFF);
    DWORD offsetHigh = (DWORD)(targetOffset >> 32);

    const char* pView = (const char*)MapViewOfFile(hMapping, FILE_MAP_READ, offsetHigh, offsetLow, 1024);
    if (pView) {
        std::cout << "Спроецировано по адресу: " << (void*)pView << std::endl;
        UnmapViewOfFile(pView);
    }

    CloseHandle(hMapping);
    CloseHandle(hFile);
    return 0;
}
Почему смещение не может быть случайным?

Диспетчер виртуальной памяти Windows отображает файлы на уровне аппаратного MMU процессора. Системная гранулярность выделения памяти (dwAllocationGranularity) составляет 64 КБ. Если передать некратное смещение (например 500 байт), функция завершится с ошибкой ERROR_MAPPED_ALIGNMENT (код 1132).

Раздел 3: 3_advanced_io_memory/

3. Прямой ввод-вывод, симуляция СУБД, сброс TLB и память ядра

Практика №1: Небуферизированный ввод-вывод (код 87 и VirtualAlloc)
task1_no_buffering_bench.cpp
Ошибка 87 (ERROR_INVALID_PARAMETER): При флаге FILE_FLAG_NO_INTERMEDIATE_BUFFERING данные идут в обход кэша Windows прямо в контроллер DMA. Адрес буфера, смещение и размер обязаны быть кратны размеру сектора диска (4096 байт). Чтение в обычный массив char buf[1000] возвращает ошибку 87!
Исправление через VirtualAlloc

Функция VirtualAlloc(NULL, 65536, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE) выделяет память у ядра, гарантированно выравнивая базовый адрес по границе страницы 4096 байт, что устраняет ошибку 87.

Практика №2: Симуляция СУБД (10 000 записей и сброс TLB)
task2_dbms_sim_filemapping_tlb.cpp
Причина просадки Режима Б (TLB Flush): При частых повторных вызовах MapViewOfFile и UnmapViewOfFile ядро Windows вынуждено непрерывно перестраивать таблицу страниц (Page Table) и вызывать инструкцию процессора INVLPG, сбрасывая аппаратный кэш трансляции адресов (TLB Flush). Стандартный I/O с кэшем ОС удерживает страницы стабильно в RAM и работает быстрее.
Практика №3: Выделение памяти напрямую у ядра Windows
task3_direct_kernel_memory.cpp
task3_direct_kernel_memory.cpp 
#include <windows.h>
#include <iostream>

int main() {
    size_t count = 10000;
    size_t sizeInBytes = count * sizeof(int);

    // 1. Выделение памяти напрямую у ядра Windows
    int* pArray = (int*)VirtualAlloc(NULL, sizeInBytes, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);

    // 2. Заполнение квадратами индексов
    for (size_t i = 0; i < count; ++i) {
        pArray[i] = (int)(i * i);
    }

    // 3. Вывод первых 10 элементов
    for (size_t i = 0; i < 10; ++i) {
        std::cout << "pArray[" << i << "] = " << pArray[i] << std::endl;
    }

    // 4. Освобождение памяти ядру (размер обязан быть 0!)
    VirtualFree(pArray, 0, MEM_RELEASE);
    return 0;
}
Построчный разбор параметров

1. MEM_RESERVE — резервирует непрерывный диапазон в виртуальном адресном пространстве процесса.

2. MEM_COMMIT — привязывает физические страницы ОЗУ к зарезервированному адресному диапазону.

3. VirtualFree(pArray, 0, MEM_RELEASE) — возвращает страницы обратно ядру. Параметр размера обязан быть равен 0.
