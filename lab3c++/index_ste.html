Задача 1: Асинхронное копирование 1 ГБ файла (OVERLAPPED + hEvent)
task1_async_copy_overlapped.cpp
Программа копирует файл асинхронно, блоками по 64 КБ. Программа читает и пишет файл через OVERLAPPED. События hReadEvent и hWriteEvent сообщают о завершении операций.
task1_async_copy_overlapped.cpp
#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <iomanip>

const DWORD CHUNK_SIZE = 64 * 1024; // 64 КБ на один блок передачи

// Функция создания тестового файла заданного размера (в мегабайтах)
bool CreateDummyFile(const wchar_t* path, DWORD sizeMB) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    LARGE_INTEGER liSize;
    liSize.QuadPart = (LONGLONG)sizeMB * 1024 * 1024;
    
    // Быстрое задание размера файла через SetFilePointerEx и SetEndOfFile
    if (!SetFilePointerEx(hFile, liSize, NULL, FILE_BEGIN) || !SetEndOfFile(hFile)) {
        CloseHandle(hFile);
        return false;
    }
    CloseHandle(hFile);
    return true;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    _setmode(_fileno(stdout), _O_U8TEXT);
    #endif

    const wchar_t* srcPath = L"source_large.dat";
    const wchar_t* dstPath = L"destination_copy.dat";

    std::wcout << L"=== АСИНХРОННОЕ КОПИРОВАНИЕ ФАЙЛА (OVERLAPPED + hEvent) ===" << std::endl;

    // Проверяем наличие исходного файла. Если нет — создаем демонстрационный файл 1024 МБ (1 ГБ)
    WIN32_FIND_DATAW findData;
    HANDLE hFind = FindFirstFileW(srcPath, &findData);
    if (hFind == INVALID_HANDLE_VALUE) {
        std::wcout << L"Создание тестового файла на 1024 МБ (1 ГБ)..." << std::endl;
        if (!CreateDummyFile(srcPath, 1024)) {
            std::wcout << L"Ошибка при создании тестового файла: " << GetLastError() << std::endl;
            return 1;
        }
        std::wcout << L"Тестовый файл успешно создан." << std::endl;
    } else {
        FindClose(hFind);
    }

    // 1. Открытие исходного файла с флагом FILE_FLAG_OVERLAPPED
    HANDLE hSrc = CreateFileW(
        srcPath,
        GENERIC_READ,
        FILE_SHARE_READ,
        NULL,
        OPEN_EXISTING,
        FILE_FLAG_OVERLAPPED,
        NULL
    );
    if (hSrc == INVALID_HANDLE_VALUE) {
        std::wcout << L"Не удалось открыть исходный файл: " << GetLastError() << std::endl;
        return 1;
    }

    // 2. Создание целевого файла с флагом FILE_FLAG_OVERLAPPED
    HANDLE hDst = CreateFileW(
        dstPath,
        GENERIC_WRITE,
        0,
        NULL,
        CREATE_ALWAYS,
        FILE_FLAG_OVERLAPPED,
        NULL
    );
    if (hDst == INVALID_HANDLE_VALUE) {
        std::wcout << L"Не удалось создать целевой файл: " << GetLastError() << std::endl;
        CloseHandle(hSrc);
        return 1;
    }

    // Получаем полный размер файла
    LARGE_INTEGER fileSize;
    GetFileSizeEx(hSrc, &fileSize);
    std::wcout << L"Размер исходного файла: " << (fileSize.QuadPart / (1024 * 1024)) << L" МБ" << std::endl;

    // 3. Создание событий для контроля завершения операций чтения и записи
    HANDLE hReadEvent = CreateEventW(NULL, TRUE, FALSE, NULL);
    HANDLE hWriteEvent = CreateEventW(NULL, TRUE, FALSE, NULL);

    char* buffer = new char[CHUNK_SIZE];

    OVERLAPPED ovRead;
    OVERLAPPED ovWrite;
    memset(&ovRead, 0, sizeof(ovRead));
    memset(&ovWrite, 0, sizeof(ovWrite));

    ovRead.hEvent = hReadEvent;
    ovWrite.hEvent = hWriteEvent;

    ULONGLONG currentOffset = 0;
    DWORD bytesTransferred = 0;

    std::wcout << L"Запуск асинхронного копирования блоками по " << (CHUNK_SIZE / 1024) << L" КБ..." << std::endl;

    while (currentOffset < (ULONGLONG)fileSize.QuadPart) {
        DWORD bytesToRead = CHUNK_SIZE;
        if (currentOffset + bytesToRead > (ULONGLONG)fileSize.QuadPart) {
            bytesToRead = (DWORD)(fileSize.QuadPart - currentOffset);
        }

        // Задаем смещение для чтения
        ovRead.Offset = (DWORD)(currentOffset & 0xFFFFFFFF);
        ovRead.OffsetHigh = (DWORD)(currentOffset >> 32);
        ResetEvent(hReadEvent);

        // Асинхронный вызов ReadFile
        BOOL readRet = ReadFile(hSrc, buffer, bytesToRead, NULL, &ovRead);
        if (!readRet && GetLastError() != ERROR_IO_PENDING) {
            std::wcout << L"Ошибка при асинхронном чтении: " << GetLastError() << std::endl;
            break;
        }

        // Ожидаем завершения чтения через событие hEvent
        WaitForSingleObject(hReadEvent, INFINITE);
        GetOverlappedResult(hSrc, &ovRead, &bytesTransferred, FALSE);

        if (bytesTransferred == 0) break;

        // Задаем смещение для записи
        ovWrite.Offset = ovRead.Offset;
        ovWrite.OffsetHigh = ovRead.OffsetHigh;
        ResetEvent(hWriteEvent);

        // Асинхронный вызов WriteFile
        BOOL writeRet = WriteFile(hDst, buffer, bytesTransferred, NULL, &ovWrite);
        if (!writeRet && GetLastError() != ERROR_IO_PENDING) {
            std::wcout << L"Ошибка при асинхронной записи: " << GetLastError() << std::endl;
            break;
        }

        // Ожидаем завершения записи через событие hEvent
        WaitForSingleObject(hWriteEvent, INFINITE);
        GetOverlappedResult(hDst, &ovWrite, &bytesTransferred, FALSE);

        currentOffset += bytesTransferred;

        // Вывод прогресса в консоль
        double percent = ((double)currentOffset / fileSize.QuadPart) * 100.0;
        std::wcout << L"\rПрогресс копирования: " << std::fixed << std::setprecision(1) << percent << L"% ("
                   << (currentOffset / (1024 * 1024)) << L" / " << (fileSize.QuadPart / (1024 * 1024)) << L" МБ)" << std::flush;
    }

    std::wcout << L"\nКопирование успешно завершено!" << std::endl;

    // Освобождение ресурсов и закрытие дескрипторов
    CloseHandle(hReadEvent);
    CloseHandle(hWriteEvent);
    CloseHandle(hSrc);
    CloseHandle(hDst);
    delete[] buffer;

    return 0;
}

Как запустить — задача 1

Только Windows. Программа использует Win32 API: windows.h, CreateFileW и OVERLAPPED. На Linux и macOS программа не компилируется и не запускается.

rem MSVC (Developer Command Prompt for VS):
cl /EHsc /W4 task1_async_copy_overlapped.cpp
rem MinGW-w64:
g++ -O2 -o task1_async_copy_overlapped.exe task1_async_copy_overlapped.cpp
rem Запуск из папки с exe (файлы создаются в текущей папке):
task1_async_copy_overlapped.exe

Файлы: Входной файл заранее создавать не нужно. Если файла source_large.dat нет в текущей папке, программа сама создаёт файл на 1024 МБ (1 ГБ) из нулей. Файл требует свободное место от 1 ГБ, а копия destination_copy.dat требует ещё 1 ГБ, то есть всего от 2 ГБ. Если файл source_large.dat уже есть, программа использует файл как есть, любого размера и содержимого, и не пересоздаёт файл. При каждом запуске программа стирает destination_copy.dat и пишет файл заново (CREATE_ALWAYS).

Ожидаемый вывод: Программа печатает заголовок. Программа печатает «Размер исходного файла: 1024 МБ». Затем программа печатает строку прогресса в одну строку: от 0,0% до 100,0%. Программа печатает «Копирование успешно завершено!». Копирование 1 ГБ блоками по 64 КБ требует 16 384 итераций. Время зависит от диска: на SSD — секунды, на HDD — дольше.

Что сломается: Без 2 ГБ свободного места создание файла даёт ошибку с кодом GetLastError(). Код 112 означает: диск переполнен. Программа выходит с кодом 1. Запуск не из своей папки создаёт файлы в другой папке, а не рядом с exe. Закрытие консоли во время работы оставляет недописанный файл destination_copy.dat. При следующем запуске программа просто переписывает файл.

Нюанс: Программа выполняет асинхронные вызовы, но ждёт завершения каждого блока через WaitForSingleObject. Поэтому чтение и запись идут строго по очереди. Это демонстрация механизма, а не ускорение копирования.

Задача 2: Чтение MFT (резидентность vs нерезидентность)
task2_mft_resident_theory.cpp
FSCTL_GET_RETRIEVAL_POINTERS показывает место $DATA для трёх файлов. Файлы имеют размер 200 байт, 2500 байт и 500 КБ. $DATA лежит внутри записи MFT или в кластерах тома.

#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <vector>

// Функция анализа размещения файла в NTFS (резидентный в MFT vs нерезидентный с экстентами)
void AnalyzeNtfsFileAllocation(const wchar_t* filePath, DWORD sizeInBytes) {
    // Создаем тестовый файл нужного размера
    HANDLE hFile = CreateFileW(
        filePath,
        GENERIC_READ | GENERIC_WRITE,
        FILE_SHARE_READ,
        NULL,
        CREATE_ALWAYS,
        FILE_ATTRIBUTE_NORMAL,
        NULL
    );

    if (hFile == INVALID_HANDLE_VALUE) {
        std::wcout << L"Ошибка открытия файла: " << GetLastError() << std::endl;
        return;
    }

    std::vector<char> buffer(sizeInBytes, 'X');
    DWORD written = 0;
    WriteFile(hFile, buffer.data(), sizeInBytes, &written, NULL);

    // Запрос цепочек кластеров (Data Runs / Extents) через FSCTL_GET_RETRIEVAL_POINTERS
    STARTING_VCN_INPUT_BUFFER inputVcn;
    inputVcn.StartingVcn.QuadPart = 0;

    BYTE outBuf[1024];
    DWORD bytesReturned = 0;

    BOOL fsctlSuccess = DeviceIoControl(
        hFile,
        FSCTL_GET_RETRIEVAL_POINTERS,
        &inputVcn,
        sizeof(inputVcn),
        outBuf,
        sizeof(outBuf),
        &bytesReturned,
        NULL
    );

    std::wcout << L"--------------------------------------------------" << std::endl;
    std::wcout << L"Файл: " << filePath << L" (Размер: " << sizeInBytes << L" байт)" << std::endl;

    if (!fsctlSuccess) {
        DWORD err = GetLastError();
        if (err == ERROR_HANDLE_EOF) {
            std::wcout << L"Статус MFT: РЕЗИДЕНТНЫЙ ФАЙЛ (Resident Attribute $DATA)" << std::endl;
            std::wcout << L"  -> Данные файла целиком хранятся внутри 1024-байтной записи MFT!" << std::endl;
            std::wcout << L"  -> Файл не занимает отдельных кластеров на дисковом томе (Extents = 0)." << std::endl;
        } else {
            std::wcout << L"Ошибка FSCTL_GET_RETRIEVAL_POINTERS: " << err << std::endl;
        }
    } else {
        RETRIEVAL_POINTERS_BUFFER* rp = reinterpret_cast<RETRIEVAL_POINTERS_BUFFER*>(outBuf);
        std::wcout << L"Статус MFT: НЕРЕЗИДЕНТНЫЙ ФАЙЛ (Non-Resident Attribute $DATA)" << std::endl;
        std::wcout << L"  -> Данные не поместились в MFT-запись и вынесены в отдельные кластеры тома." << std::endl;
        std::wcout << L"  -> Количество цепочек кластеров (Extents): " << rp->ExtentCount << std::endl;
        for (DWORD i = 0; i < rp->ExtentCount; ++i) {
            std::wcout << L"     Цепочка " << i << L": LCN = " << rp->Extents[i].Lcn.QuadPart 
                       << L", NextVCN = " << rp->Extents[i].NextVcn.QuadPart << std::endl;
        }
    }

    CloseHandle(hFile);
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    _setmode(_fileno(stdout), _O_U8TEXT);
    #endif

    std::wcout << L"=== АНАЛИЗ ЧТЕНИЯ MFT: РЕЗИДЕНТНЫЕ И НЕРЕЗИДЕНТНЫЕ ДАННЫЕ ===" << std::endl;

    // Сравнение 1: Маленький файл (200 байт — строго помещается в MFT)
    AnalyzeNtfsFileAllocation(L"test_resident_200b.dat", 200);

    // Сравнение 2: Большой файл (2500 байт — не помещается в 1 КБ запись MFT)
    AnalyzeNtfsFileAllocation(L"test_nonresident_2500b.dat", 2500);

    // Сравнение 3: Файл объемом 500 КБ (512 000 байт — нерезидентный массив данных)
    AnalyzeNtfsFileAllocation(L"test_large_500kb.dat", 500 * 1024);

    return 0;
}

Задача 3: Реальный физический размер файлов вне таблицы MFT
task3_mft_physical_size.cpp
Старый вариант программы печатал 200 и 2000. Эти числа — логический размер, а не занятое место на диске. GetCompressedFileSizeW возвращает логический размер для обычных (несжатых) файлов. Программа теперь берёт размер напрямую из AllocationSize. Файл 200 байт занимает 0 байт. Файл 2000 байт занимает 4096 байт.

#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>

void investigateFile(const wchar_t* fileName, DWORD bytesToWrite) {
    HANDLE hFile = CreateFileW(
        fileName,
        GENERIC_READ | GENERIC_WRITE,
        0,
        NULL,
        CREATE_ALWAYS,
        FILE_ATTRIBUTE_NORMAL,
        NULL
    );

    if (hFile == INVALID_HANDLE_VALUE) {
        std::wcout << L"Ошибка создания файла: " << GetLastError() << std::endl;
        return;
    }

    char* buffer = new char[bytesToWrite];
    memset(buffer, 'A', bytesToWrite);

    DWORD bytesWritten = 0;
    WriteFile(hFile, buffer, bytesToWrite, &bytesWritten, NULL);
    delete[] buffer;

    // ПРЯМОЙ ПУТЬ: спрашиваем у самой NTFS, сколько места выделено под файл.
    // FILE_STANDARD_INFO.AllocationSize — это реальный выделенный размер:
    // 0 для резидентного файла (данные лежат внутри записи MFT, кластеры не
    // выделялись), для нерезидентного — число кластеров × размер кластера.
    // Запрос делается по открытому хэндлу, до CloseHandle.
    FILE_STANDARD_INFO stdInfo = {};
    LONGLONG physicalSize = -1;
    if (GetFileInformationByHandleEx(hFile, FileStandardInfo, &stdInfo, sizeof(stdInfo))) {
        physicalSize = stdInfo.AllocationSize.QuadPart;
    } else {
        std::wcout << L"Ошибка GetFileInformationByHandleEx: " << GetLastError() << std::endl;
    }

    CloseHandle(hFile);

    // ДЛЯ СРАВНЕНИЯ: ловушка GetCompressedFileSizeW.
    // Несмотря на имя, для обычного (несжатого, не разреженного) файла она
    // возвращает ЛОГИЧЕСКИЙ размер, как GetFileSize. Так написано в самой
    // документации Microsoft: "If the file is not compressed or a sparse
    // file, the value obtained is the actual file size, the same as the value
    // returned by a call to GetFileSize". Реальный размер на диске она даёт
    // только для сжатых и разреженных файлов. Поэтому старый вариант этой
    // программы печатал 200 и 2000 вместо правильных 0 и 4096.
    DWORD highSize = 0;
    DWORD compressedSize = GetCompressedFileSizeW(fileName, &highSize);
    if (compressedSize == INVALID_FILE_SIZE && GetLastError() != NO_ERROR) {
        std::wcout << L"Ошибка GetCompressedFileSizeW: " << GetLastError() << std::endl;
    }

    std::wcout << L"Файл: " << fileName << std::endl;
    std::wcout << L"  Записано данных (логический размер): " << bytesToWrite << L" байт." << std::endl;
    std::wcout << L"  Занято на диске (физический размер, напрямую): " << physicalSize << L" байт." << std::endl;
    std::wcout << L"  GetCompressedFileSizeW (для сравнения): " << compressedSize << L" байт." << std::endl;
    std::wcout << L"--------------------------------------------------" << std::endl;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    _setmode(_fileno(stdout), _O_U8TEXT);
    #endif

    std::wcout << L"=== ИССЛЕДОВАНИЕ ФИЗИЧЕСКОГО РАЗМЕРА ФАЙЛА ВНЕ ТАБЛИЦЫ MFT ===" << std::endl;

    investigateFile(L"large_file.txt", 2000); // Кейс 2: 2000 байт (нерезидентный)

    return 0;
}
