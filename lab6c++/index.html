task1_createprocess_notepad.cpp 
#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 1: ЖИЗНЕННЫЙ ЦИКЛ ПРОЦЕССА И ОЧИСТКА РЕСУРСОВ ЯДРА ===" << std::endl;
    std::cout << "Запускаем Блокнот (notepad.exe) через CreateProcess..." << std::endl;

    STARTUPINFO si;
    PROCESS_INFORMATION pi;

    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));

    si.cb = sizeof(si);

    wchar_t cmdLine[] = L"notepad.exe";

    BOOL ok = CreateProcess(
        NULL,           // lpApplicationName
        cmdLine,        // lpCommandLine
        NULL,           // lpProcessAttributes
        NULL,           // lpThreadAttributes
        FALSE,          // bInheritHandles
        0,              // dwCreationFlags
        NULL,           // lpEnvironment
        NULL,           // lpCurrentDirectory
        &si,            // lpStartupInfo
        &pi             // lpProcessInformation
    );

    if (!ok) {
        std::cout << "Ошибка CreateProcess: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Блокнот запущен. PID процесса: " << pi.dwProcessId
              << ", PID главного потока: " << pi.dwThreadId << std::endl;
    std::cout << "Ждём, пока вы закроете Блокнот..." << std::endl;

    DWORD waitResult = WaitForSingleObject(pi.hProcess, INFINITE);

    if (waitResult == WAIT_OBJECT_0) {
        std::cout << "Блокнот закрыт пользователем." << std::endl;
    } else {
        std::cout << "WaitForSingleObject вернул неожиданный код: " << waitResult
                  << " (GetLastError: " << GetLastError() << ")" << std::endl;
    }

    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены." << std::endl;

    return 0;
}

#include <windows.h>Подключает Win32 API: CreateProcess, WaitForSingleObject, CloseHandle, STARTUPINFO, PROCESS_INFORMATION, ZeroMemory, GetLastError.Без этого заголовка ни одна функция Windows недоступна — это главный API-заголовок для системного программирования.
#include <iostream>Подключает std::cout / std::endl для вывода в консоль.Вывод нужен для наблюдения за жизненным циклом: запуск, ожидание, закрытие.
int main() {Точка входа консольного приложения.Здесь живёт весь сценарий: запуск → ожидание → очистка.
#if defined(_WIN32) system("chcp 65001 > nul"); #endifПереключает кодовую страницу консоли на UTF-8 (65001), чтобы русский текст не превращался в кракозябры.Директива препроцессора: команда выполняется только под Windows; на других системах код просто не компилируется в программу.
std::cout << "=== ЗАДАНИЕ 1 ..."Печатает заголовок программы.Маркер в выводе: видно, какая программа запущена и что она делает.
std::cout << "Запускаем Блокнот..."Сообщает пользователю о начале запуска.Пользователь понимает, что сейчас откроется окно Блокнота — это не случайный процесс.
STARTUPINFO si;Объявляет структуру с параметрами запуска: внешний вид окна, дескрипторы стандартных потоков.Без неё CreateProcess не знает, как показать окно нового процесса.
PROCESS_INFORMATION pi;Объявляет структуру для результатов: сюда система запишет дескрипторы процесса и потока и их ID.Это наш единственный способ получить дескриптор запущенного процесса — без него нельзя ни ждать его, ни закрыть.
ZeroMemory(&si, sizeof(si));Заполняет всю структуру si нулями.В локальных переменных — мусор из стека. Случайный бит во флагах STARTUPINFO может заставить CreateProcess вести себя непредсказуемо.
ZeroMemory(&pi, sizeof(pi));То же обнуление для структуры результатов.Защита от мусора: при ошибке CreateProcess поля pi останутся нулями, а не случайными значениями.
si.cb = sizeof(si);Записывает в поле cb размер структуры в байтах.CreateProcess по этому полю определяет версию структуры. Если cb равен 0 или мусору — вызов завершится ошибкой ERROR_INVALID_PARAMETER.
wchar_t cmdLine[] = L"notepad.exe";Создаёт изменяемый буфер с командной строкой запуска Блокнота (широкие символы, Unicode).Документация требует именно изменяемый буфер: CreateProcess имеет право менять строку на месте. Строковый литерал напрямую передавать нельзя — он в памяти только для чтения.
BOOL ok = CreateProcess( ... );Создаёт новый процесс notepad.exe и возвращает TRUE при успехе, FALSE при ошибке.Это центральный вызов задания: только здесь рождается дочерний процесс. Результат обязательно проверяется.
NULL, // lpApplicationNameИмя исполняемого файла не задано отдельно.Когда первый параметр NULL, система берёт имя программы из командной строки — в нашем случае "notepad.exe".
cmdLine, // lpCommandLineПередаёт командную строку с именем программы.Система ищет notepad.exe по правилам поиска исполняемых файлов (каталог приложения, системные каталоги, PATH). Блокнот есть в любой Windows — путь указывать не нужно.
NULL, // lpProcessAttributesАтрибуты безопасности процесса — по умолчанию.Дескриптор процесса нельзя будет унаследовать дочерними процессами; для задания это не нужно.
NULL, // lpThreadAttributesАтрибуты безопасности главного потока — по умолчанию.То же, что и для процесса: наследование дескриптора потока не требуется.
FALSE, // bInheritHandlesЗапрещает Блокноту наследовать дескрипторы нашей программы.Наследование открыло бы Блокноту доступ к нашим файлам и объектам. Здесь это не нужно и небезопасно.
0, // dwCreationFlagsБез особых флагов создания.Процесс создаётся обычным образом: не приостановленным, не в отдельной консоли, не с повышенными правами.
NULL, // lpEnvironmentБлок окружения для потомка не задан явно.NULL означает: потомок получает копию блока окружения родителя. Для Блокнота это нормально; механизм копии подробно разбирается в задании 2.
NULL, // lpCurrentDirectoryРабочий каталог потомка не задан явно.NULL означает: потомок наследует текущий каталог родителя. Блокноту каталог не важен.
&si, // lpStartupInfoПередаёт заполненную структуру запуска.Система читает отсюда, как показать окно Блокнота (по умолчанию — обычное окно).
&pi // lpProcessInformationПередаёт адрес структуры для результатов.Сюда система запишет hProcess, hThread, dwProcessId, dwThreadId — всё, что нужно для ожидания и очистки.
if (!ok) { ... return 1; }Если CreateProcess вернул FALSE — печатает код ошибки через GetLastError() и завершает программу с кодом 1.Без этой проверки программа продолжила бы ждать по мусорному дескриптору. Код 1 сигнализирует системе, что запуск не удался.
std::cout << "Блокнот запущен. PID процесса: " << pi.dwProcessId ...Выводит ID процесса и ID его главного потока.PID — доказательство, что процесс реально создан; по нему его можно найти в Диспетчере задач.
std::cout << "Ждём, пока вы закроете Блокнот..."Объясняет, почему консоль «зависла».Без этой строки ожидание выглядит как зависание программы; пользователь должен знать, что нужно закрыть Блокнот.
DWORD waitResult = WaitForSingleObject(pi.hProcess, INFINITE);Блокирует поток до тех пор, пока дескриптор процесса не станет сигнальным, то есть пока Блокнот не завершится. Ждёт без ограничения времени.Это и есть «дождаться закрытия»: дескриптор процесса переходит в сигнальное состояние ровно в момент его завершения. INFINITE — пользователь может держать Блокнот открытым сколь угодно долго.
if (waitResult == WAIT_OBJECT_0) { ... }Проверяет, что ожидание завершилось именно завершением процесса, а не ошибкой.WAIT_OBJECT_0 — нормальный исход. Любой другой код означает сбой ожидания, и его надо показать, а не молча идти дальше.
std::cout << "Блокнот закрыт пользователем."Подтверждает, что ожидание закончилось.Пользователь видит, что программа заметила закрытие Блокнота.
else { std::cout << "WaitForSingleObject вернул неожиданный код..." }Ветка ошибки ожидания: печатает код возврата и GetLastError().Честная диагностика вместо молчания: если ожидание сорвалось, причина должна быть видна.
CloseHandle(pi.hProcess);Закрывает дескриптор процесса Блокнота.КРИТИЧЕСКОЕ УСЛОВИЕ ЗАДАНИЯ. Пока дескриптор открыт, ядро хранит объект процесса и его код завершения в памяти, даже после завершения процесса. Незакрытый дескриптор — утечка ресурса ядра и занятая запись в таблице дескрипторов.
CloseHandle(pi.hThread);Закрывает дескриптор главного потока Блокнота.CreateProcess возвращает ДВА дескриптора, и закрыть нужно ОБА. Забытый hThread — та же утечка, что и забытый hProcess. Порядок закрытия значения не имеет.
std::cout << "Дескрипторы ... закрыты ..."Подтверждает выполнение очистки.На защите видно, что программа дошла до CloseHandle, а не завершилась раньше.
return 0;Успешное завершение программы с кодом 0.Код 0 — общепринятый признак нормального завершения; его видит вызывающая среда.
Как запустить
  1. Файлов заранее создавать не нужно: notepad.exe — стандартная утилита, она есть в любой Windows.
  2. Скомпилировать. MSVC (Developer Command Prompt): cl /EHsc task1_createprocess_notepad.cpp — получится task1_createprocess_notepad.exe. MinGW: g++ -o task1.exe task1_createprocess_notepad.cpp.
  3. Запустить exe из консоли. Откроется Блокнот; консоль будет ждать.
  4. Закрыть Блокнот (крестик или Alt+F4). Консоль выведет сообщение о закрытии и о закрытых дескрипторах, затем завершится.
Ожидаемый вывод: строка о запуске с PID процесса и потока → «Ждём, пока вы закроете Блокнот...» → после закрытия Блокнота «Блокнот закрыт пользователем.» → «Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены.»
Если закрыть консоль раньше Блокнота — Блокнот продолжит работать: завершение родителя не убивает дочерний процесс. Это нормально: процессы в Windows независимы.

#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 2: PARENT (РОДИТЕЛЬ) ===" << std::endl;

    const wchar_t* varName = L"STUDENT_ID";
    const wchar_t* valueBefore = L"19045";

    if (!SetEnvironmentVariable(varName, valueBefore)) {
        std::cout << "Ошибка SetEnvironmentVariable: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Parent: установлена переменная STUDENT_ID=";
    std::wcout << valueBefore << std::endl;

    STARTUPINFO si;
    PROCESS_INFORMATION pi;
    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));
    si.cb = sizeof(si);

    wchar_t cmdLine[] = L"Child.exe";

    BOOL ok = CreateProcess(
        NULL,           // lpApplicationName
        cmdLine,        // lpCommandLine
        NULL,           // lpProcessAttributes
        NULL,           // lpThreadAttributes
        FALSE,          // bInheritHandles
        0,              // dwCreationFlags
        NULL,           // lpEnvironment — копия блока окружения родителя
        NULL,           // lpCurrentDirectory
        &si,            // lpStartupInfo
        &pi             // lpProcessInformation
    );

    if (!ok) {
        std::cout << "Ошибка CreateProcess (не найден Child.exe рядом с Parent.exe?): "
                  << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Parent: Child.exe запущен. PID потомка: " << pi.dwProcessId << std::endl;

    const wchar_t* valueAfter = L"99999";
    if (!SetEnvironmentVariable(varName, valueAfter)) {
        std::cout << "Ошибка повторного SetEnvironmentVariable: " << GetLastError() << std::endl;
    } else {
        std::cout << "Parent: переменная у СЕБЯ перезаписана на STUDENT_ID=";
        std::wcout << valueAfter << std::endl;
        std::cout << "Parent: у потомка осталась КОПИЯ со значением 19045 — она не изменилась." << std::endl;
    }

    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Parent: дескрипторы потомка закрыты, родитель завершает работу." << std::endl;

    return 0;
}

CloseHandle(pi.hProcess);Закрывает дескриптор процесса-потомка.Родитель не ждёт потомка, но дескриптор всё равно надо закрыть: иначе объект процесса будет висеть в ядре как утечка (см. задание 1). Закрытие дескриптора НЕ завершает потомка.
CloseHandle(pi.hThread);Закрывает дескриптор главного потока потомка.Оба дескриптора из PROCESS_INFORMATION закрываются всегда, даже если ожидание не использовалось.
std::cout << "Parent: дескрипторы потомка закрыты..."Сообщает о завершении очистки.Видно, что родитель дошёл до конца сценария и ничего не забыл.
return 0;Родитель мгновенно завершается с кодом 0.Это шаг 3 задания: родитель уходит сразу, а потомок продолжает жить сам — завершение родителя потомка не убивает.
Как запустить — Parent.exe
  1. Скомпилировать ОБА файла отдельно. MSVC: cl /EHsc task2_parent.cpp → task2_parent.exe, затем переименовать в Parent.exe; cl /EHsc task2_child.cpp → переименовать в Child.exe. MinGW: g++ -o Parent.exe task2_parent.cpp и g++ -o Child.exe task2_child.cpp.
  2. Положить Parent.exe и Child.exe в ОДНУ папку. Это обязательное условие: Parent ищет Child.exe по относительному имени рядом с собой.
  3. Запустить Parent.exe (двойной клик или из консоли). Откроется окно Child.exe, а окно Parent закроется почти сразу.
Если Parent перезапишет переменную ДО CreateProcess — потомок получит уже новое значение 99999, и эксперимент потеряет смысл: доказать автономность будет нечем. Порядок «сначала CreateProcess, потом перезапись» — суть задания.
Если Child.exe лежит в другой папке — CreateProcess вернёт FALSE, Parent напечатает ошибку с кодом GetLastError (обычно 2 — ERROR_FILE_NOT_FOUND) и завершится с кодом 1.

#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 2: CHILD (ПОТОМОК) ===" << std::endl;

    const wchar_t* varName = L"STUDENT_ID";
    wchar_t buffer[256];

    DWORD len = GetEnvironmentVariable(varName, buffer, sizeof(buffer) / sizeof(buffer[0]));

    if (len == 0) {
        DWORD err = GetLastError();
        if (err == ERROR_ENVVAR_NOT_FOUND) {
            std::cout << "ОШИБКА: переменная STUDENT_ID не найдена в блоке окружения." << std::endl;
            std::cout << "Скорее всего, Child.exe запущен напрямую, без Parent.exe," << std::endl;
            std::cout << "поэтому никто не установил эту переменную перед запуском." << std::endl;
        } else {
            std::cout << "Ошибка GetEnvironmentVariable: " << err << std::endl;
        }
    } else {
        std::cout << "Child: прочитана переменная STUDENT_ID=";
        std::wcout << buffer << std::endl;

        if (wcscmp(buffer, L"19045") == 0) {
            std::cout << "Child: значение равно 19045 — оно НЕ изменилось вслед за родителем." << std::endl;
            std::cout << "Child: доказано — Environment Block копируется при создании процесса," << std::endl;
            std::cout << "Child: потомок живёт со своим снимком окружения, а не с общей памятью." << std::endl;
        } else {
            std::cout << "Child: ВНИМАНИЕ: значение отличается от ожидаемого 19045!" << std::endl;
            std::cout << "Child: возможно, Child.exe запущен не из-под Parent.exe," << std::endl;
            std::cout << "Child: а из среды, где STUDENT_ID уже была задана иначе." << std::endl;
        }
    }

    std::cout << "Child: родитель уже завершён, а потомок продолжает работу сам." << std::endl;
    system("pause");

    return 0;
}

Как запустить — Child.exe и ожидаемый вывод
  1. Файлов заранее создавать не нужно. Компиляция — см. карточку Parent: g++ -o Child.exe task2_child.cpp (MinGW) или cl /EHsc task2_child.cpp (MSVC) с переименованием в Child.exe.
  2. Обычный сценарий — запуск через Parent.exe (см. выше): оба exe в одной папке, запускается Parent. Ожидаемый вывод Child: STUDENT_ID=19045 → «значение равно 19045 — оно НЕ изменилось вслед за родителем» → пауза до нажатия клавиши.
  3. Отдельный запуск: Child.exe можно запустить и без Parent — двойным кликом или из консоли. Тогда переменная STUDENT_ID в его окружении не задана, и Child выведет: «ОШИБКА: переменная STUDENT_ID не найдена в блоке окружения» и объяснение, что его запустили напрямую без Parent.exe. Это нормальное поведение, а не баг.
