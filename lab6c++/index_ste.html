Задание 1: Жизненный цикл процесса и очистка ресурсов ядра
task1_createprocess_notepad.cpp
task1_createprocess_notepad.cpp 
#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 1: ЖИЗНЕННЫЙ ЦИКЛ ПРОЦЕССА И ОЧИСТКА РЕСУРСОВ ЯДРА ===" << std::endl;
    std::cout << "Запускаем Блокнот (notepad.exe) через CreateProcess..." << std::endl;

    STARTUPINFO si;
    PROCESS_INFORMATION pi;

    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));

    si.cb = sizeof(si);

    wchar_t cmdLine[] = L"notepad.exe";

    BOOL ok = CreateProcess(
        NULL,           // lpApplicationName
        cmdLine,        // lpCommandLine
        NULL,           // lpProcessAttributes
        NULL,           // lpThreadAttributes
        FALSE,          // bInheritHandles
        0,              // dwCreationFlags
        NULL,           // lpEnvironment
        NULL,           // lpCurrentDirectory
        &si,            // lpStartupInfo
        &pi             // lpProcessInformation
    );

    if (!ok) {
        std::cout << "Ошибка CreateProcess: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Блокнот запущен. PID процесса: " << pi.dwProcessId
              << ", PID главного потока: " << pi.dwThreadId << std::endl;
    std::cout << "Ждём, пока вы закроете Блокнот..." << std::endl;

    DWORD waitResult = WaitForSingleObject(pi.hProcess, INFINITE);

    if (waitResult == WAIT_OBJECT_0) {
        std::cout << "Блокнот закрыт пользователем." << std::endl;
    } else {
        std::cout << "WaitForSingleObject вернул неожиданный код: " << waitResult
                  << " (GetLastError: " << GetLastError() << ")" << std::endl;
    }

    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены." << std::endl;

    return 0;
}

#include <windows.h>Заголовок даёт программе функции Windows: CreateProcess, WaitForSingleObject, CloseHandle, STARTUPINFO, PROCESS_INFORMATION, ZeroMemory, GetLastError.Без заголовка программа не вызывает ни одну функцию Windows. Заголовок — главный API-заголовок для системного программирования.
#include <iostream>Заголовок даёт программе std::cout и std::endl для вывода в консоль.Вывод показывает жизненный цикл: запуск, ожидание, закрытие.
int main() {Функция main — точка входа программы.Функция main запускает процесс, ждёт его завершения и закрывает дескрипторы.

STARTUPINFO si;Объявление создаёт структуру si. Структура хранит параметры запуска: вид окна и дескрипторы стандартных потоков.Без структуры si функция CreateProcess не знает, как показать окно нового процесса.
PROCESS_INFORMATION pi;Объявление создаёт структуру pi. Структура принимает результаты: дескрипторы процесса и потока и их ID.Только структура pi даёт программе дескриптор запущенного процесса. Без дескриптора программа не ждёт процесс и не закрывает его.
ZeroMemory(&si, sizeof(si));Функция заполняет структуру si нулями.Локальные переменные хранят мусор из стека. Случайный бит во флагах меняет поведение функции CreateProcess. Ноль убирает мусор.
ZeroMemory(&pi, sizeof(pi));Функция заполняет структуру pi нулями.При ошибке функции CreateProcess поля pi остаются нулями, а не мусором. Ноль защищает программу от случайных значений.
si.cb = sizeof(si);Команда записывает в поле cb размер структуры в байтах.Функция CreateProcess читает поле cb и узнаёт версию структуры. Поле cb со значением 0 или мусором даёт ошибку ERROR_INVALID_PARAMETER.
wchar_t cmdLine[] = L"notepad.exe";Объявление создаёт изменяемый буфер cmdLine. Буфер хранит командную строку «notepad.exe» в широких символах (Unicode).Документация требует изменяемый буфер: функция CreateProcess может менять строку на месте. Строковый литерал лежит в памяти только для чтения. Программа не передаёт литерал напрямую.
BOOL ok = CreateProcess( ... );Функция создаёт новый процесс notepad.exe. Функция возвращает TRUE при успехе и FALSE при ошибке.Это главный вызов задания: только здесь рождается потомок. Программа всегда проверяет результат вызова.
NULL, // lpApplicationNameПараметр не задаёт имя исполняемого файла отдельно.Первый параметр равен NULL. Система берёт имя программы из командной строки: «notepad.exe».
cmdLine, // lpCommandLineПараметр передаёт командную строку с именем программы.Система ищет файл notepad.exe в каталоге приложения, в системных каталогах и в PATH. Блокнот есть в каждой Windows. Путь указывать не нужно.
NULL, // lpProcessAttributesПараметр задаёт атрибуты безопасности процесса по умолчанию.Потомки не наследуют дескриптор процесса. Задание не требует наследования.
NULL, // lpThreadAttributesПараметр задаёт атрибуты безопасности главного потока по умолчанию.Потомки не наследуют дескриптор потока. Задание не требует наследования.
FALSE, // bInheritHandlesПараметр запрещает Блокноту наследовать дескрипторы программы.Наследование открывает Блокноту доступ к файлам и объектам программы. Задание не требует наследования. Запрет защищает программу.
0, // dwCreationFlagsПараметр задаёт ноль особых флагов создания.Процесс создаётся обычным образом: не приостановленным, не в отдельной консоли, без повышенных прав.
NULL, // lpEnvironmentПараметр не задаёт блок окружения для потомка.Значение NULL даёт потомку копию блока окружения родителя. Для Блокнота копия подходит. Задание 2 разбирает механизм копии подробно.
NULL, // lpCurrentDirectoryПараметр не задаёт рабочий каталог потомка.Значение NULL даёт потомку текущий каталог родителя. Блокноту каталог не важен.
&si, // lpStartupInfoПараметр передаёт заполненную структуру запуска.Система читает структуру и показывает окно Блокнота обычным образом.
&pi // lpProcessInformationПараметр передаёт адрес структуры для результатов.Система записывает сюда hProcess, hThread, dwProcessId и dwThreadId. Эти значения нужны для ожидания и очистки.
if (!ok) { ... return 1; }При значении FALSE программа печатает код ошибки из GetLastError(). Программа завершается с кодом 1.Без проверки программа ждёт по мусорному дескриптору. Код 1 сообщает системе: запуск не удался.
std::cout << "Блокнот запущен. PID процесса: " << pi.dwProcessId ...Команда выводит ID процесса и ID его главного потока.PID доказывает: процесс реально создан. Пользователь находит процесс в Диспетчере задач по PID.
std::cout << "Ждём, пока вы закроете Блокнот..."Команда объясняет, почему консоль ждёт.Без строки ожидание выглядит как зависание программы. Пользователь знает: нужно закрыть Блокнот.
DWORD waitResult = WaitForSingleObject(pi.hProcess, INFINITE);Функция останавливает поток, пока Блокнот не завершится. Дескриптор процесса становится сигнальным ровно при завершении процесса. Параметр INFINITE снимает ограничение времени.Это и есть ожидание закрытия. Пользователь держит Блокнот открытым сколько угодно. Функция ждёт без ограничения времени.
if (waitResult == WAIT_OBJECT_0) { ... }Проверка подтверждает: ожидание завершилось завершением процесса, а не ошибкой.Код WAIT_OBJECT_0 — нормальный исход. Другой код означает сбой ожидания. Программа показывает сбой, а не молчит о нём.
std::cout << "Блокнот закрыт пользователем."Команда подтверждает закрытие Блокнота.Пользователь видит: программа заметила закрытие Блокнота.
else { std::cout << "WaitForSingleObject вернул неожиданный код..." }Ветка ошибки печатает код возврата и код из GetLastError().Честная диагностика показывает причину сбоя. Программа не молчит при сбое ожидания.
CloseHandle(pi.hProcess);Функция закрывает дескриптор процесса Блокнота.Это критическое условие задания. Пока дескриптор открыт, ядро хранит объект процесса и код завершения. Незакрытый дескриптор — утечка ресурса ядра.
CloseHandle(pi.hThread);Функция закрывает дескриптор главного потока Блокнота.Функция CreateProcess возвращает два дескриптора. Программа закрывает оба. Забытый дескриптор потока — та же утечка. Порядок закрытия не важен.
std::cout << "Дескрипторы ... закрыты ..."Команда подтверждает выполнение очистки.На защите видно: программа дошла до CloseHandle и не завершилась раньше.
return 0;Программа завершается с кодом 0.Код 0 означает нормальное завершение. Вызывающая среда читает код 0.
Как запустить
  1. Не создавайте файлы заранее. Утилита notepad.exe есть в каждой Windows.
  2. Скомпилируйте файл. MSVC: выполните команду cl /EHsc task1_createprocess_notepad.cpp. Команда создаёт файл task1_createprocess_notepad.exe. MinGW: выполните команду g++ -o task1.exe task1_createprocess_notepad.cpp.
  3. Запустите exe из консоли. Блокнот открывается. Консоль ждёт закрытия Блокнота.
  4. Закройте Блокнот: нажмите крестик или клавиши Alt+F4. Консоль показывает сообщение о закрытии и о закрытых дескрипторах. Затем программа завершается.
Ожидаемый вывод. Программа показывает строку о запуске с PID процесса и потока. Затем программа показывает строку «Ждём, пока вы закроете Блокнот...». После закрытия Блокнота программа показывает строку «Блокнот закрыт пользователем.». Затем программа показывает строку «Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены.».
Закрытие консоли раньше Блокнота не закрывает Блокнот. Завершение родителя не завершает потомка. Процессы в Windows независимы. Это нормальное поведение.

task2_parent.cpp 
#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 2: PARENT (РОДИТЕЛЬ) ===" << std::endl;

    const wchar_t* varName = L"STUDENT_ID";
    const wchar_t* valueBefore = L"19045";

    if (!SetEnvironmentVariable(varName, valueBefore)) {
        std::cout << "Ошибка SetEnvironmentVariable: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Parent: установлена переменная STUDENT_ID=";
    std::wcout << valueBefore << std::endl;

    STARTUPINFO si;
    PROCESS_INFORMATION pi;
    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));
    si.cb = sizeof(si);

    wchar_t cmdLine[] = L"Child.exe";

    BOOL ok = CreateProcess(
        NULL,           // lpApplicationName
        cmdLine,        // lpCommandLine
        NULL,           // lpProcessAttributes
        NULL,           // lpThreadAttributes
        FALSE,          // bInheritHandles
        0,              // dwCreationFlags
        NULL,           // lpEnvironment — копия блока окружения родителя
        NULL,           // lpCurrentDirectory
        &si,            // lpStartupInfo
        &pi             // lpProcessInformation
    );

    if (!ok) {
        std::cout << "Ошибка CreateProcess (не найден Child.exe рядом с Parent.exe?): "
                  << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Parent: Child.exe запущен. PID потомка: " << pi.dwProcessId << std::endl;

    const wchar_t* valueAfter = L"99999";
    if (!SetEnvironmentVariable(varName, valueAfter)) {
        std::cout << "Ошибка повторного SetEnvironmentVariable: " << GetLastError() << std::endl;
    } else {
        std::cout << "Parent: переменная у СЕБЯ перезаписана на STUDENT_ID=";
        std::wcout << valueAfter << std::endl;
        std::cout << "Parent: у потомка осталась КОПИЯ со значением 19045 — она не изменилась." << std::endl;
    }

    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Parent: дескрипторы потомка закрыты, родитель завершает работу." << std::endl;

    return 0;
}

&si, // lpStartupInfoПараметр передаёт параметры запуска.Система читает параметры и настраивает окно потомка.
&pi // lpProcessInformationПараметр передаёт структуру для результатов.Структура даёт dwProcessId потомка и дескрипторы hProcess и hThread.
if (!ok) { ... return 1; }При ошибке запуска программа печатает текст и код из GetLastError(). Программа завершается с кодом 1.Частая причина ошибки: файл Child.exe не лежит рядом с Parent.exe. Сообщение прямо подсказывает причину.
std::cout << "Parent: Child.exe запущен. PID потомка: " << pi.dwProcessIdКоманда выводит PID запущенного потомка.PID доказывает: потомок реально создан. Пользователь находит потомка в Диспетчере задач по PID.
const wchar_t* valueAfter = L"99999";Объявление задаёт новое значение переменной: 99999.Новое значение отличается от 19045. Разница видна сразу. Константа хранит значение в одном месте.
if (!SetEnvironmentVariable(varName, valueAfter)) { ... }Функция перезаписывает переменную STUDENT_ID на 99999 в своём блоке окружения. Ветка else печатает два факта: у родителя стало 99999, у потомка осталась копия 19045.Это шаг 3 задания. Порядок решает всё: перезапись идёт строго после вызова CreateProcess. Перезапись до вызова дала бы потомку уже 99999. Эксперимент потерял бы смысл.
CloseHandle(pi.hProcess);Функция закрывает дескриптор процесса-потомка.Родитель не ждёт потомка. Дескриптор всё равно нужно закрыть. Иначе объект процесса висит в ядре как утечка. Закрытие дескриптора не завершает потомка.
CloseHandle(pi.hThread);Функция закрывает дескриптор главного потока потомка.Программа закрывает оба дескриптора из структуры PROCESS_INFORMATION всегда, даже без ожидания.
std::cout << "Parent: дескрипторы потомка закрыты..."Команда сообщает о завершении очистки.Видно: родитель дошёл до конца сценария и ничего не забыл.
return 0;Родитель завершается сразу с кодом 0.Это шаг 3 задания. Родитель уходит сразу. Потомок продолжает жить сам. Завершение родителя не завершает потомка.
Как запустить — Parent.exe
  1. Скомпилируйте оба файла отдельно. MSVC: выполните команду cl /EHsc task2_parent.cpp. Команда создаёт файл task2_parent.exe. Переименуйте файл в Parent.exe. Выполните команду cl /EHsc task2_child.cpp. Переименуйте файл в Child.exe. MinGW: выполните команду g++ -o Parent.exe task2_parent.cpp. Выполните команду g++ -o Child.exe task2_child.cpp.
  2. Положите файлы Parent.exe и Child.exe в одну папку. Это обязательное условие. Родитель ищет файл Child.exe по относительному имени рядом с собой.
  3. Запустите файл Parent.exe двойным щелчком или из консоли. Откроется окно Child.exe. Окно Parent закроется почти сразу.
Не перезаписывайте переменную до вызова CreateProcess. Потомок получит уже новое значение 99999. Эксперимент потеряет смысл. Доказать автономность будет нечем. Порядок решает всё: сначала CreateProcess, потом перезапись.
Файл Child.exe в другой папке даёт ошибку. Функция CreateProcess возвращает FALSE. Родитель печатает ошибку с кодом из GetLastError. Обычно код равен 2 (ERROR_FILE_NOT_FOUND). Родитель завершается с кодом 1.

#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 2: CHILD (ПОТОМОК) ===" << std::endl;

    const wchar_t* varName = L"STUDENT_ID";
    wchar_t buffer[256];

    DWORD len = GetEnvironmentVariable(varName, buffer, sizeof(buffer) / sizeof(buffer[0]));

    if (len == 0) {
        DWORD err = GetLastError();
        if (err == ERROR_ENVVAR_NOT_FOUND) {
            std::cout << "ОШИБКА: переменная STUDENT_ID не найдена в блоке окружения." << std::endl;
            std::cout << "Скорее всего, Child.exe запущен напрямую, без Parent.exe," << std::endl;
            std::cout << "поэтому никто не установил эту переменную перед запуском." << std::endl;
        } else {
            std::cout << "Ошибка GetEnvironmentVariable: " << err << std::endl;
        }
    } else {
        std::cout << "Child: прочитана переменная STUDENT_ID=";
        std::wcout << buffer << std::endl;

        if (wcscmp(buffer, L"19045") == 0) {
            std::cout << "Child: значение равно 19045 — оно НЕ изменилось вслед за родителем." << std::endl;
            std::cout << "Child: доказано — Environment Block копируется при создании процесса," << std::endl;
            std::cout << "Child: потомок живёт со своим снимком окружения, а не с общей памятью." << std::endl;
        } else {
            std::cout << "Child: ВНИМАНИЕ: значение отличается от ожидаемого 19045!" << std::endl;
            std::cout << "Child: возможно, Child.exe запущен не из-под Parent.exe," << std::endl;
            std::cout << "Child: а из среды, где STUDENT_ID уже была задана иначе." << std::endl;
        }
    }

    std::cout << "Child: родитель уже завершён, а потомок продолжает работу сам." << std::endl;
    system("pause");

    return 0;
}
