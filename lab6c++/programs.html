#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 1: ЖИЗНЕННЫЙ ЦИКЛ ПРОЦЕССА И ОЧИСТКА РЕСУРСОВ ЯДРА ===" << std::endl;
    std::cout << "Запускаем Блокнот (notepad.exe) через CreateProcess..." << std::endl;

    // 1. Структуры, описывающие запуск нового процесса.
    //    STARTUPINFO — как должен выглядеть новый процесс (окно, дескрипторы ввода/вывода).
    //    PROCESS_INFORMATION — что вернёт нам система о запущенном процессе.
    STARTUPINFO si;
    PROCESS_INFORMATION pi;

    // 2. Обязательно обнуляем обе структуры: в них не должно остаться мусора.
    //    Неинициализированные поля STARTUPINFO — классическая причина
    //    непонятных сбоев CreateProcess.
    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));

    // 3. Поле cb — размер структуры в байтах. CreateProcess проверяет его,
    //    чтобы понять, какую версию STARTUPINFO мы используем.
    si.cb = sizeof(si);

    // 4. Создаём дочерний процесс — Блокнот.
    //    Первый параметр (lpApplicationName) — NULL: имя программы берётся
    //    из командной строки. Второй параметр — командная строка.
    //    Остальные параметры по умолчанию: наследование дескрипторов выключено,
    //    флаги создания — 0, окружение и текущий каталог — как у родителя.
    //    Важно: буфер командной строки должен быть изменяемым (wchar_t[]),
    //    потому что CreateProcess имеет право его менять.
    wchar_t cmdLine[] = L"notepad.exe";

    BOOL ok = CreateProcess(
        NULL,           // lpApplicationName: NULL — имя берём из командной строки
        cmdLine,        // lpCommandLine: изменяемая строка "notepad.exe"
        NULL,           // lpProcessAttributes: дескриптор процесса не наследуется
        NULL,           // lpThreadAttributes: дескриптор потока не наследуется
        FALSE,          // bInheritHandles: дочерний процесс НЕ наследует наши дескрипторы
        0,              // dwCreationFlags: без особых флагов
        NULL,           // lpEnvironment: NULL — копия блока окружения родителя
        NULL,           // lpCurrentDirectory: NULL — текущий каталог родителя
        &si,            // lpStartupInfo: как запускать (заполнена выше)
        &pi             // lpProcessInformation: сюда система запишет дескрипторы и ID
    );

    // 5. Проверяем результат. FALSE — процесс не создан, деталей нет:
    //    ни дескрипторов, ни PID — закрывать нечего, просто выходим с кодом 1.
    if (!ok) {
        std::cout << "Ошибка CreateProcess: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Блокнот запущен. PID процесса: " << pi.dwProcessId
              << ", PID главного потока: " << pi.dwThreadId << std::endl;
    std::cout << "Ждём, пока вы закроете Блокнот..." << std::endl;

    // 6. Ждём завершения дочернего процесса. WaitForSingleObject блокирует
    //    наш поток, пока объект-дескриптор процесса pi.hProcess не перейдёт
    //    в сигнальное состояние — это происходит ровно в момент завершения
    //    процесса. INFINITE — ждать без ограничения по времени.
    //    Возврат WAIT_OBJECT_0 означает: процесс завершился.
    DWORD waitResult = WaitForSingleObject(pi.hProcess, INFINITE);

    if (waitResult == WAIT_OBJECT_0) {
        std::cout << "Блокнот закрыт пользователем." << std::endl;
    } else {
        std::cout << "WaitForSingleObject вернул неожиданный код: " << waitResult
                  << " (GetLastError: " << GetLastError() << ")" << std::endl;
    }

    // 7. КРИТИЧЕСКОЕ УСЛОВИЕ ЗАДАНИЯ: закрываем дескрипторы процесса и потока.
    //    Пока дескриптор открыт, ядро хранит объект процесса (и его код
    //    завершения) в памяти, даже если сам процесс уже завершён.
    //    Незакрытый дескриптор = утечка ресурса ядра + занятая запись
    //    в таблице дескрипторов процесса. Порядок закрытия не важен,
    //    но закрыть нужно ОБА дескриптора.
    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены." << std::endl;

    return 0;
}

#include <windows.h>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 1: ЖИЗНЕННЫЙ ЦИКЛ ПРОЦЕССА И ОЧИСТКА РЕСУРСОВ ЯДРА ===" << std::endl;
    std::cout << "Запускаем Блокнот (notepad.exe) через CreateProcess..." << std::endl;

    STARTUPINFO si;
    PROCESS_INFORMATION pi;

    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));

    si.cb = sizeof(si);

    wchar_t cmdLine[] = L"notepad.exe";

    BOOL ok = CreateProcess(
        NULL,
        cmdLine,
        NULL,
        NULL,
        FALSE,
        0,
        NULL,
        NULL,
        &si,
        &pi
    );

    if (!ok) {
        std::cout << "Ошибка CreateProcess: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Блокнот запущен. PID процесса: " << pi.dwProcessId
              << ", PID главного потока: " << pi.dwThreadId << std::endl;
    std::cout << "Ждём, пока вы закроете Блокнот..." << std::endl;

    DWORD waitResult = WaitForSingleObject(pi.hProcess, INFINITE);

    if (waitResult == WAIT_OBJECT_0) {
        std::cout << "Блокнот закрыт пользователем." << std::endl;
    } else {
        std::cout << "WaitForSingleObject вернул неожиданный код: " << waitResult
                  << " (GetLastError: " << GetLastError() << ")" << std::endl;
    }

    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены." << std::endl;

    return 0;
}
Как запустить
  1. Файлов заранее создавать не нужно: notepad.exe — стандартная утилита, она есть в любой Windows.
  2. Скомпилировать. MSVC (Developer Command Prompt): cl /EHsc task1_createprocess_notepad.cpp — получится task1_createprocess_notepad.exe. MinGW: g++ -o task1.exe task1_createprocess_notepad.cpp.
  3. Запустить exe из консоли. Откроется Блокнот; консоль будет ждать.
  4. Закрыть Блокнот (крестик или Alt+F4). Консоль выведет сообщение о закрытии и о закрытых дескрипторах, затем завершится.
Ожидаемый вывод: строка о запуске с PID процесса и потока → «Ждём, пока вы закроете Блокнот...» → после закрытия Блокнота «Блокнот закрыт пользователем.» → «Дескрипторы процесса и потока закрыты (CloseHandle). Ресурсы ядра освобождены.»
Если закрыть консоль раньше Блокнота — Блокнот продолжит работать: завершение родителя не убивает дочерний процесс. Это нормально: процессы в Windows независимы.

#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 2: PARENT (РОДИТЕЛЬ) ===" << std::endl;

    // 1. Динамически формируем переменную окружения STUDENT_ID=19045
    //    в СОБСТВЕННОМ блоке окружения этого процесса.
    //    SetEnvironmentVariable меняет переменную только в памяти текущего
    //    процесса — ни в системе, ни в других процессах она не появляется.
    const wchar_t* varName = L"STUDENT_ID";
    const wchar_t* valueBefore = L"19045";

    if (!SetEnvironmentVariable(varName, valueBefore)) {
        std::cout << "Ошибка SetEnvironmentVariable: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Parent: установлена переменная STUDENT_ID=";
    std::wcout << valueBefore << std::endl;

    // 2. Готовим структуры для запуска дочернего процесса — Child.exe.
    //    Обнуление обязательно: мусор в STARTUPINFO ломает CreateProcess.
    STARTUPINFO si;
    PROCESS_INFORMATION pi;
    ZeroMemory(&si, sizeof(si));
    ZeroMemory(&pi, sizeof(pi));
    si.cb = sizeof(si);

    // 3. Запускаем Child.exe. Имя — относительное: L"Child.exe".
    //    Поэтому Child.exe ОБЯЗАТЕЛЬНО должен лежать в той же папке,
    //    что и Parent.exe, иначе CreateProcess вернёт FALSE.
    //    lpEnvironment = NULL — это ключевой момент задания: потомок
    //    получает КОПИЮ блока окружения родителя на момент запуска.
    //    Копию — значит, потомок живёт со своим снимком, а не с общей памятью.
    wchar_t cmdLine[] = L"Child.exe";

    BOOL ok = CreateProcess(
        NULL,           // lpApplicationName: NULL — имя берём из командной строки
        cmdLine,        // lpCommandLine: "Child.exe" (должен лежать рядом с Parent.exe)
        NULL,           // lpProcessAttributes: дескриптор процесса не наследуется
        NULL,           // lpThreadAttributes: дескриптор потока не наследуется
        FALSE,          // bInheritHandles: дочерний процесс НЕ наследует наши дескрипторы
        0,              // dwCreationFlags: без особых флагов
        NULL,           // lpEnvironment: NULL — Child получает КОПИЮ нашего блока окружения
        NULL,           // lpCurrentDirectory: NULL — текущий каталог родителя
        &si,            // lpStartupInfo
        &pi             // lpProcessInformation: дескрипторы и ID потомка
    );

    if (!ok) {
        std::cout << "Ошибка CreateProcess (не найден Child.exe рядом с Parent.exe?): "
                  << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Parent: Child.exe запущен. PID потомка: " << pi.dwProcessId << std::endl;

    // 4. СРАЗУ ПОСЛЕ ЗАПУСКА перезаписываем переменную У СЕБЯ на другое значение.
    //    Это происходит ПОСЛЕ CreateProcess, поэтому копия блока окружения,
    //    уже переданная потомку, НЕ меняется: у потомка остался снимок
    //    со значением 19045, а у нас теперь 99999.
    const wchar_t* valueAfter = L"99999";
    if (!SetEnvironmentVariable(varName, valueAfter)) {
        std::cout << "Ошибка повторного SetEnvironmentVariable: " << GetLastError() << std::endl;
    } else {
        std::cout << "Parent: переменная у СЕБЯ перезаписана на STUDENT_ID=";
        std::wcout << valueAfter << std::endl;
        std::cout << "Parent: у потомка осталась КОПИЯ со значением 19045 — она не изменилась." << std::endl;
    }

    // 5. Родитель НЕ ждёт потомка (в отличие от задания 1 здесь нет
    //    WaitForSingleObject) — он мгновенно завершается. Потомок при этом
    //    продолжает жить сам по себе: завершение родителя его не убивает.
    //    Но перед выходом всё равно закрываем дескрипторы потомка,
    //    иначе они утекут в ядро (см. задание 1).
    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    std::cout << "Parent: дескрипторы потомка закрыты, родитель завершает работу." << std::endl;

    return 0;
}

Как запустить — Parent.exe
  1. Скомпилировать ОБА файла отдельно. MSVC: cl /EHsc task2_parent.cpp → task2_parent.exe, затем переименовать в Parent.exe; cl /EHsc task2_child.cpp → переименовать в Child.exe. MinGW: g++ -o Parent.exe task2_parent.cpp и g++ -o Child.exe task2_child.cpp.
  2. Положить Parent.exe и Child.exe в ОДНУ папку. Это обязательное условие: Parent ищет Child.exe по относительному имени рядом с собой.
  3. Запустить Parent.exe (двойной клик или из консоли). Откроется окно Child.exe, а окно Parent закроется почти сразу.
Если Parent перезапишет переменную ДО CreateProcess — потомок получит уже новое значение 99999, и эксперимент потеряет смысл: доказать автономность будет нечем. Порядок «сначала CreateProcess, потом перезапись» — суть задания.
Если Child.exe лежит в другой папке — CreateProcess вернёт FALSE, Parent напечатает ошибку с кодом GetLastError (обычно 2 — ERROR_FILE_NOT_FOUND) и завершится с кодом 1.

#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ЗАДАНИЕ 2: CHILD (ПОТОМОК) ===" << std::endl;

    // 1. Читаем переменную STUDENT_ID из СОБСТВЕННОГО блока окружения
    //    этого процесса. Блок окружения потомок получил от родителя
    //    в момент CreateProcess — это КОПИЯ (снимок) блока родителя,
    //    а не общая память. Поэтому то, что родитель делает со своей
    //    переменной ПОСЛЕ запуска, на нас уже никак не влияет.
    const wchar_t* varName = L"STUDENT_ID";
    wchar_t buffer[256];

    //    GetEnvironmentVariable возвращает длину значения (без завершающего
    //    нуля). 0 означает: переменная не найдена (или её значение — пустая
    //    строка; различают по GetLastError() == ERROR_ENVVAR_NOT_FOUND).
    DWORD len = GetEnvironmentVariable(varName, buffer, sizeof(buffer) / sizeof(buffer[0]));

    if (len == 0) {
        DWORD err = GetLastError();
        if (err == ERROR_ENVVAR_NOT_FOUND) {
            std::cout << "ОШИБКА: переменная STUDENT_ID не найдена в блоке окружения." << std::endl;
            std::cout << "Скорее всего, Child.exe запущен напрямую, без Parent.exe," << std::endl;
            std::cout << "поэтому никто не установил эту переменную перед запуском." << std::endl;
        } else {
            std::cout << "Ошибка GetEnvironmentVariable: " << err << std::endl;
        }
    } else {
        // 2. Переменная найдена — выводим её значение.
        std::cout << "Child: прочитана переменная STUDENT_ID=";
        std::wcout << buffer << std::endl;

        // 3. Доказательство автономности: сравниваем со значением,
        //    которое установил родитель ДО запуска (19045).
        //    Родитель после запуска перезаписал её у себя на 99999,
        //    но наша копия не изменилась — процессы изолированы.
        if (wcscmp(buffer, L"19045") == 0) {
            std::cout << "Child: значение равно 19045 — оно НЕ изменилось вслед за родителем." << std::endl;
            std::cout << "Child: доказано — Environment Block копируется при создании процесса," << std::endl;
            std::cout << "Child: потомок живёт со своим снимком окружения, а не с общей памятью." << std::endl;
        } else {
            std::cout << "Child: ВНИМАНИЕ: значение отличается от ожидаемого 19045!" << std::endl;
            std::cout << "Child: возможно, Child.exe запущен не из-под Parent.exe," << std::endl;
            std::cout << "Child: а из среды, где STUDENT_ID уже была задана иначе." << std::endl;
        }
    }

    // 4. Ждём нажатия клавиши, чтобы окно консоли не закрылось раньше,
    //    чем пользователь увидит вывод. Без этой паузы потомок завершился
    //    бы мгновенно — особенно если его запустили двойным кликом.
    //    system("pause") выводит "Для продолжения нажмите любую клавишу . . ."
    //    и ждёт нажатия. Родитель к этому моменту уже завершён —
    //    потомок продолжает работу самостоятельно.
    std::cout << "Child: родитель уже завершён, а потомок продолжает работу сам." << std::endl;
    system("pause");

    return 0;
}

Как запустить — Child.exe и ожидаемый вывод
  1. Файлов заранее создавать не нужно. Компиляция — см. карточку Parent: g++ -o Child.exe task2_child.cpp (MinGW) или cl /EHsc task2_child.cpp (MSVC) с переименованием в Child.exe.
  2. Обычный сценарий — запуск через Parent.exe (см. выше): оба exe в одной папке, запускается Parent. Ожидаемый вывод Child: STUDENT_ID=19045 → «значение равно 19045 — оно НЕ изменилось вслед за родителем» → пауза до нажатия клавиши.
  3. Отдельный запуск: Child.exe можно запустить и без Parent — двойным кликом или из консоли. Тогда переменная STUDENT_ID в его окружении не задана, и Child выведет: «ОШИБКА: переменная STUDENT_ID не найдена в блоке окружения» и объяснение, что его запустили напрямую без Parent.exe. Это нормальное поведение, а не баг.
Ожидаемый вывод полного сценария. Окно Parent: «установлена переменная STUDENT_ID=19045» → «Child.exe запущен. PID потомка: ...» → «переменная у СЕБЯ перезаписана на STUDENT_ID=99999» → «у потомка осталась КОПИЯ со значением 19045» → окно Parent закрывается. Окно Child: «прочитана переменная STUDENT_ID=19045» → «значение равно 19045 — оно НЕ изменилось вслед за родителем» → пауза.
