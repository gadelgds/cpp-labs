#include <iostream>
#include <iomanip>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== НАИЗУСТЬ: ЯВНОЕ ПРИВЕДЕНИЕ ТИПОВ ЧЕРЕЗ static_cast ===" << std::endl;

    // ЧАСТЬ 1: Вычитание int из long с явным приведением к long
    long valLong = 100000L;
    int valInt = 25000;

    // Вычитание с явным приведением static_cast<long>()
    long resultLong = valLong - static_cast<long>(valInt);

    std::cout << "\n[Целочисленное вычитание]:" << std::endl;
    std::cout << "  Переменная long:                 " << valLong << std::endl;
    std::cout << "  Переменная int:                  " << valInt << std::endl;
    std::cout << "  Результат (long - static_cast<long>(int)): " << resultLong << std::endl;

    // ЧАСТЬ 2: Вычитание float из double с явным приведением к double
    double valDouble = 125.75;
    float valFloat = 25.25f;

    // Вычитание с явным приведением static_cast<double>()
    double resultDouble = valDouble - static_cast<double>(valFloat);

    std::cout << "\n[Вещественное вычитание с плавающей точкой]:" << std::endl;
    std::cout << std::fixed << std::setprecision(4);
    std::cout << "  Переменная double:               " << valDouble << std::endl;
    std::cout << "  Переменная float:                " << valFloat << std::endl;
    std::cout << "  Результат (double - static_cast<double>(float)): " << resultDouble << std::endl;

    return 0;
}

Задача 2 (Самостоятельно): Указатели void* и разрядность x64

#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== САМОСТОЯТЕЛЬНО 1: УКАЗАТЕЛИ void* И РАЗМЕРЫ В x64 ===" << std::endl;

    // 1. Создание переменных типов int и long со значениями
    int numInt = 42;
    long numLong = 123456L;

    // 2. Создание независимых универсальных указателей типа void*
    void* ptrVoid1 = &numInt;
    void* ptrVoid2 = &numLong;

    // 3. Обратное преобразование указателей void* в исходные типы int* и long*
    int* ptrInt = static_cast<int*>(ptrVoid1);
    long* ptrLong = static_cast<long*>(ptrVoid2);

    // 4. Извлечение значений из памяти (разыменование) и вывод на экран
    std::cout << "\n[Извлеченные значения]:" << std::endl;
    std::cout << "  Значение из ptrInt:  " << *ptrInt << " (адрес: " << ptrInt << ")" << std::endl;
    std::cout << "  Значение из ptrLong: " << *ptrLong << " (адрес: " << ptrLong << ")" << std::endl;

    // 5. Определение размеров указателей и типов в архитектуре сборки (x64)
    std::cout << "\n[Размеры типов и указателей в текущей системе]:" << std::endl;
    std::cout << "  Разрядность сборки:    " << (sizeof(void*) * 8) << "-бит" << std::endl;
    std::cout << "  sizeof(int):           " << sizeof(int) << " байт (" << sizeof(int) * 8 << " бит)" << std::endl;
    std::cout << "  sizeof(long):          " << sizeof(long) << " байт (" << sizeof(long) * 8 << " бит)" << std::endl;
    std::cout << "  sizeof(long long):     " << sizeof(long long) << " байт (" << sizeof(long long) * 8 << " бит)" << std::endl;
    std::cout << "  sizeof(void*):         " << sizeof(void*) << " байт (" << sizeof(void*) * 8 << " бит)" << std::endl;
    std::cout << "  sizeof(int*):          " << sizeof(int*) << " байт (" << sizeof(int*) * 8 << " бит)" << std::endl;
    std::cout << "  sizeof(long*):         " << sizeof(long*) << " байт (" << sizeof(long*) * 8 << " бит)" << std::endl;

    return 0;
}

Главный подвох: *ptrVoid1 написать нельзя. Компилятор запрещает разыменование void*. У void нет размера. Компилятор не знает, сколько байт читать. Сначала сделай static_cast к конкретному типу. Только потом ставь *.
Итог: все указатели в x64 занимают по 8 байт. Тип данных не влияет на размер указателя. Сами типы различаются: int и long в Windows занимают по 4 байта. long long занимает 8 байт.

=== САМОСТОЯТЕЛЬНО 1: УКАЗАТЕЛИ void* И РАЗМЕРЫ В x64 === [Извлеченные значения]: Значение из ptrInt: 42 (адрес: 0x000000A1B2C3D4E5) Значение из ptrLong: 123456 (адрес: 0x000000A1B2C3D4E9) [Размеры типов и указателей в текущей системе]: Разрядность сборки: 64-бит sizeof(int): 4 байт (32 бит) sizeof(long): 4 байт (32 бит) sizeof(long long): 8 байт (64 бит) sizeof(void*): 8 байт (64 бит) sizeof(int*): 8 байт (64 бит) sizeof(long*): 8 байт (64 бит)

Типичные ошибки:

  • Сборка в 32-битном режиме (цель x86 или -m32): sizeof(void*) равен 4. Разрядность равна 32 битам. Это не ошибка программы. Это другая архитектура сборки. Задача требует именно x64.
  • sizeof(long) равен 8: сборка выполнена под Linux (модель LP64). В Windows long всегда занимает 4 байта.
  • Ошибка компиляции int* p = ptrVoid1;: так и должно быть. Неявное приведение void* в int* запрещено. Нужен static_cast.
  • Ошибка компиляции при *ptrVoid1: разыменование void* запрещено стандартом. Сначала приведи указатель к конкретному типу.

#include <windows.h>
#include <iostream>
#include <string>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== САМОСТОЯТЕЛЬНО 2: ПРОЕЦИРОВАНИЕ ФАЙЛА С ФИКСИРОВАННЫМ СМЕЩЕНИЕМ ===" << std::endl;

    const wchar_t* filename = L"mapped_large_test.txt";

    // 1. Получаем информацию об архитектуре и гранулярности распределения памяти Windows
    SYSTEM_INFO sysInfo;
    GetSystemInfo(&sysInfo);
    DWORD allocGranularity = sysInfo.dwAllocationGranularity; // Обычно 65536 байт (64 КБ)

    std::cout << "Системная гранулярность выделения памяти (Allocation Granularity): "
              << allocGranularity << " байт (" << (allocGranularity / 1024) << " КБ)" << std::endl;

    // 2. Создаем тестовый файл достаточного размера (например, 256 КБ = 4 * 64 КБ)
    HANDLE hFile = CreateFileW(
        filename,
        GENERIC_READ | GENERIC_WRITE,
        FILE_SHARE_READ,
        NULL,
        CREATE_ALWAYS,
        FILE_ATTRIBUTE_NORMAL,
        NULL
    );

    if (hFile == INVALID_HANDLE_VALUE) {
        std::cout << "Ошибка создания файла: " << GetLastError() << std::endl;
        return 1;
    }

    // Заполняем файл данными: 4 блока по 64 КБ с уникальными маркерами
    const DWORD totalFileSize = 4 * allocGranularity; // 256 КБ
    std::string fillData(totalFileSize, '.');

    // Размещаем маркер во 2-м блоке (смещение 2 * allocGranularity = 128 КБ)
    std::string secretMessage = "[СЕКРЕТНЫЙ ТЕКСТ НА СМЕЩЕНИИ 128 КБ]";
    size_t targetOffset = 2 * allocGranularity;
    memcpy(&fillData[targetOffset], secretMessage.c_str(), secretMessage.length());

    DWORD bytesWritten = 0;
    WriteFile(hFile, fillData.data(), totalFileSize, &bytesWritten, NULL);

    // 3. Создаем объект проекции файла (File Mapping Object)
    HANDLE hMapping = CreateFileMappingW(
        hFile,
        NULL,
        PAGE_READONLY,
        0,
        0,
        NULL
    );

    if (hMapping == NULL) {
        std::cout << "Ошибка CreateFileMapping: " << GetLastError() << std::endl;
        CloseHandle(hFile);
        return 1;
    }

    // 4. Смещение для MapViewOfFile ОБЯЗАНО быть строго кратным allocGranularity!
    // Демонстрируем расчет смещения:
    DWORD offsetLow = (DWORD)(targetOffset & 0xFFFFFFFF);
    DWORD offsetHigh = (DWORD)(((ULONGLONG)targetOffset >> 32) & 0xFFFFFFFF);
    SIZE_T viewSize = 1024; // Отображаем окно размером 1024 байта

    std::cout << "\nЗапрос проецирования участка файла:" << std::endl;
    std::cout << "  Смещение: " << targetOffset << " байт (кратно " << allocGranularity << ")" << std::endl;
    std::cout << "  Размер проецируемого окна: " << viewSize << " байт" << std::endl;

    // 5. Проецируем срез файла в виртуальную память процесса
    const char* pView = static_cast<const char*>(MapViewOfFile(
        hMapping,
        FILE_MAP_READ,
        offsetHigh,
        offsetLow,
        viewSize
    ));

    if (pView == NULL) {
        std::cout << "Ошибка MapViewOfFile: " << GetLastError() << std::endl;
    } else {
        std::cout << "  Успешно спроецировано по виртуальному адресу: " << (void*)pView << std::endl;

        // Читаем данные из памяти спроецированного окна
        std::string readChunk(pView, secretMessage.length());
        std::cout << "  Прочитанные данные: \"" << readChunk << "\"" << std::endl;

        // Отмонтируем проекцию
        UnmapViewOfFile(pView);
        std::cout << "Окно проекции успешно размонтировано (UnmapViewOfFile)." << std::endl;
    }

    // 6. Закрытие дескрипторов
    CloseHandle(hMapping);
    CloseHandle(hFile);

    return 0;
}

// 5. Проецируем срез файла в виртуальную память процессаКомментарий помечает шаг 5: отображение.Это центральный вызов программы.
const char* pView = static_cast<const char*>(MapViewOfFile( hMapping, FILE_MAP_READ, offsetHigh, offsetLow, viewSize ));Функция отображает 1024 байта файла со смещения 131072 в память процесса. Функция возвращает указатель на начало окна. Аргументы: дескриптор секции; FILE_MAP_READ; старшая и младшая части смещения; размер окна. Результат приводится к const char*.После вызова байты файла читаются как обычная память: pView[0] — первый байт маркера. Тип const char* выбран потому, что данные — текст. Слово const запрещает запись. Приведение static_cast обязательно: неявное приведение void* в const char* запрещено в C++.
if (pView == NULL) {Программа проверяет, удалось ли спроецировать окно.Смещение не кратно 64 КБ: функция возвращает NULL. GetLastError даёт 1132 (ERROR_MAPPED_ALIGNMENT). Смещение и окно выходят за конец файла: тоже NULL, но с другим кодом.
std::cout << "Ошибка MapViewOfFile: " << GetLastError() << std::endl;Программа печатает код ошибки проецирования.По коду различают причину. Код 1132 означает неверное выравнивание смещения. Другие коды означают выход за конец файла или нехватку памяти.
} else {Начинается ветка успеха: проекция создана.Разделение на ошибку и успех — стандартный паттерн Win32. Дальше код работает только с живым отображением.
std::cout << " Успешно спроецировано по виртуальному адресу: " << (void*)pView << std::endl;Программа печатает виртуальный адрес начала окна. Приведение (void*)pView заставляет cout печатать адрес, а не содержимое.Без приведения к void* вывод показал бы текст маркера вместо адреса. Это классическая ловушка char* в потоках. Адрес при каждом запуске разный из-за ASLR. Это нормально.
// Читаем данные из памяти спроецированного окнаКомментарий помечает чтение из отображения.Чтение идёт из памяти, а не через ReadFile. В этом смысл проецирования.
std::string readChunk(pView, secretMessage.length());Программа создаёт строку из байтов по адресу pView. Длина равна длине маркера. Конструктор string(const char*, size_t) читает по длине, а не до нуля.Так читается именно маркер: он начинается на смещении 128 КБ. Длина известна заранее. Конструктор по длине важен: после маркера идут точки '.', а не '\0'. Чтение до нуля прочитало бы лишнее.
std::cout << " Прочитанные данные: \"" << readChunk << "\"" << std::endl;Программа печатает маркер в кавычках: [СЕКРЕТНЫЙ ТЕКСТ НА СМЕЩЕНИИ 128 КБ].Это главное доказательство работы программы. Текст, записанный на шаге 2, прочитан обратно через память. Неверное смещение дало бы другой текст или ошибку.
// Отмонтируем проекциюКомментарий помечает освобождение отображения.Комментарий структурирует завершение работы.
UnmapViewOfFile(pView);Функция снимает отображение: освобождает диапазон виртуальных адресов окна. Результат здесь не проверяется.Каждое MapViewOfFile требует парного UnmapViewOfFile. Иначе диапазон останется занятым до конца процесса. После вызова указатель pView недействителен.
std::cout << "Окно проекции успешно размонтировано (UnmapViewOfFile)." << std::endl;Программа печатает подтверждение размонтирования.Вывод фиксирует: ресурс освобождён корректно.
}Скобка закрывает блок else.Это граница ветки успеха.
// 6. Закрытие дескрипторовКомментарий помечает шаг 6: финальная очистка.Это последний шаг программы.
CloseHandle(hMapping);Программа закрывает дескриптор секции.Порядок важен: сначала секция, потом файл. Объект ядра уничтожается при закрытии последнего дескриптора. Отображение уже снято, поэтому закрытие безопасно.
CloseHandle(hFile);Программа закрывает дескриптор файла.После закрытия другие программы открывают файл без ограничений. Дескриптор не утекает. Файл остаётся на диске: маркер можно проверить поиском.
return 0;Программа возвращает системе код 0.Код 0 возвращается даже при неудаче MapViewOfFile. Ветка ошибки лишь печатает сообщение. Это слабое место: код возврата не отличает неудачу от успеха. Для учебного кода это допустимо.
}Скобка закрывает main.Программа завершается. Локальные объекты уничтожаются.
Почему смещение обязано быть кратно 64 КБ: Windows отображает файл страницами. Границы областей памяти выравниваются по гранулярности 65536 байт. Смещение 131072 равно 2 × 65536: оно допустимо. Смещение 131073 или 100000 недопустимо: MapViewOfFile возвращает NULL. GetLastError даёт 1132 (ERROR_MAPPED_ALIGNMENT).
Итог: программа сама создаёт файл 256 КБ. Программа кладёт маркер ровно на 128 КБ. Программа проецирует окно 1 КБ с этого смещения. Программа читает маркер из памяти. Программа не вызывает ReadFile ни разу.
Как запустить — задача 3

Только Windows. Программа использует Win32 API: windows.h, CreateFileW, MapViewOfFile. Под Linux и macOS она не соберётся.

Файл заранее создавать не нужно. Программа сама создаёт mapped_large_test.txt в рабочей папке. Флаг CREATE_ALWAYS создаёт файл при каждом запуске. Программа записывает 256 КБ данных и кладёт маркер на смещение 128 КБ. Старый файл перезаписывается заново. Это нормальное поведение, а не ошибка.

Единственное условие: право на запись в рабочую папку. Без прав программа печатает «Ошибка создания файла: 5». Код 5 означает отказ в доступе. Программа завершается с кодом 1.

Компиляция (Windows):

:: MSVC — в Developer Command Prompt for VS: cl /EHsc /std:c++17 task3_memory_mapped_offset.cpp :: MinGW-w64 (g++) — в PowerShell / cmd: g++ -std=c++17 -o task3.exe task3_memory_mapped_offset.cpp

Запуск из папки с exe. Файл mapped_large_test.txt появится рядом с exe.

.\task3.exe

Ожидаемый вывод. Виртуальный адрес при каждом запуске разный. Это нормально:

=== САМОСТОЯТЕЛЬНО 2: ПРОЕЦИРОВАНИЕ ФАЙЛА С ФИКСИРОВАННЫМ СМЕЩЕНИЕМ === Системная гранулярность выделения памяти (Allocation Granularity): 65536 байт (64 КБ) Запрос проецирования участка файла: Смещение: 131072 байт (кратно 65536) Размер проецируемого окна: 1024 байт Успешно спроецировано по виртуальному адресу: 0x0000021A3B4C5000 Прочитанные данные: "[СЕКРЕТНЫЙ ТЕКСТ НА СМЕЩЕНИИ 128 КБ]" Окно проекции успешно размонтировано (UnmapViewOfFile).

Проверка файла после запуска. Рядом с exe появится mapped_large_test.txt размером 262144 байта. Маркер проверяется поиском по файлу или командой PowerShell.

:: Размер файла и проверка маркера на смещении 128 КБ: powershell -c "(Get-Item mapped_large_test.txt).Length" powershell -c "$b=[System.IO.File]::ReadAllBytes('mapped_large_test.txt'); [System.Text.Encoding]::UTF8.GetString($b[131072..131120])"

Типичные ошибки:

  • Ошибка MapViewOfFile: 1132. Смещение не кратно 64 КБ. В коде смещение 131072 = 2 × 65536. При неизменённом коде ошибки нет. Ошибка появится, если поменять targetOffset на некруглое число. Например, 100000.
  • Ошибка создания файла: 5. Нет прав на запись в папку запуска. Например, exe лежит в C:\Program Files. Запускай программу из обычной папки. Например, из папки проекта или Desktop.
  • Ошибка MapViewOfFile с другим кодом. Смещение и окно выходят за конец файла. В коде файл 256 КБ. Окно заканчивается на 129 КБ. Это меньше 256 КБ. При неизменённом коде ошибки нет. Ошибка появится, если уменьшить totalFileSize или увеличить targetOffset.
  • Нарушение доступа при чтении дальше 1024 байт от pView. Окно спроецировано только на 1 КБ. Байты за его пределами недоступны. Читай только внутри viewSize.
  • В старой версии шпаргалки код использовал OPEN_EXISTING. Это была сокращённая версия. В реальном файле стоит CREATE_ALWAYS. Файл создаётся самой программой. Для режима OPEN_EXISTING файл пришлось бы создавать заранее. Например, командой fsutil file createnew mapped_large_test.txt 262144. К актуальному коду это не относится.
