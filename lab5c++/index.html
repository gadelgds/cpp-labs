#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <vector>
#include <random>

const DWORD FILE_SIZE_10MB = 10 * 1024 * 1024; // 10 МБ
const DWORD SECTOR_SIZE = 4096;                // 4 КБ (размер сектора диска)

// Создание тестового файла на 10 МБ со случайными символами
bool GenerateRandomFile(const wchar_t* path, DWORD sizeBytes) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    const DWORD chunkSize = 64 * 1024;
    std::vector<char> buffer(chunkSize);
    std::mt19937 rng(42);
    for (size_t i = 0; i < chunkSize; ++i) buffer[i] = static_cast<char>('A' + (rng() % 26));

    DWORD writtenTotal = 0;
    while (writtenTotal < sizeBytes) {
        DWORD toWrite = min(chunkSize, sizeBytes - writtenTotal);
        DWORD written = 0;
        WriteFile(hFile, buffer.data(), toWrite, &written, NULL);
        writtenTotal += written;
    }
    CloseHandle(hFile);
    return true;
}

// Замер времени чтения через QueryPerformanceCounter
double BenchmarkReading(const wchar_t* path, DWORD flags, void* buffer, DWORD readChunkSize) {
    HANDLE hFile = CreateFileW(path, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, flags, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return -1.0;

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&start);

    DWORD bytesRead = 0;
    DWORD totalRead = 0;
    while (totalRead < FILE_SIZE_10MB) {
        if (!ReadFile(hFile, buffer, readChunkSize, &bytesRead, NULL) || bytesRead == 0) {
            break;
        }
        totalRead += bytesRead;
    }

    QueryPerformanceCounter(&end);
    CloseHandle(hFile);

    return static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
}

    // 4. Сравнение скорости чтения с кэшированием и без кэширования
    std::wcout << L"\n4. Сравнение производительности чтения 10 МБ (QueryPerformanceCounter):" << std::endl;

    // Тест А: Со стандартным кэшированием ОС (FILE_ATTRIBUTE_NORMAL)
    double timeCached = BenchmarkReading(testFile, FILE_ATTRIBUTE_NORMAL, alignedBuffer, alignedChunkSize);
    std::wcout << L"   А) Стандартное чтение (с системным кэшем):  " << timeCached << L" мс" << std::endl;

    std::wcout << L"   Б) Прямое чтение (без промежуточного кэша): " << timeDirect << L" мс" << std::endl;

    // Освобождение памяти
    VirtualFree(alignedBuffer, 0, MEM_RELEASE);

    return 0;
}

Причина просадки Режима Б (TLB Flush): При частых вызовах UnmapViewOfFile ядро Windows вынуждено непрерывно перестраивать таблицу страниц (Page Table) процесса и инициировать процессорную инструкцию INVLPG, сбрасывая аппаратный кэш трансляции адресов (TLB Flush). Стандартный I/O с системным кэшем удерживает страницы стабильно в RAM и работает значительно быстрее.

const DWORD FILE_SIZE_16MB = 16 * 1024 * 1024; // 16 МБ
const int NUM_OPERATIONS = 10000;              // 10 000 случайных модификаций

// Создание 16 МБ бинарного файла последовательности int
bool CreateDbBinaryFile(const wchar_t* path) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    const size_t intsPerBlock = 64 * 1024;
    std::vector<int> block(intsPerBlock);
    for (size_t i = 0; i < intsPerBlock; ++i) block[i] = static_cast<int>(i);

    DWORD totalInts = FILE_SIZE_16MB / sizeof(int);
    DWORD writtenInts = 0;
    while (writtenInts < totalInts) {
        DWORD toWriteInts = min(intsPerBlock, totalInts - writtenInts);
        DWORD writtenBytes = 0;
        WriteFile(hFile, block.data(), toWriteInts * sizeof(int), &writtenBytes, NULL);
        writtenInts += toWriteInts;
    }
    CloseHandle(hFile);
    return true;
}

#include <iostream>

    #if defined(_WIN32)

    #endif

    std::cout << "=== ПРАКТИКА №3: ВЫДЕЛЕНИЕ ПАМЯТИ НАПРЯМУЮ У ЯДРА WINDOWS (VirtualAlloc) ===" << std::endl;

    // 1. Запрос размера массива у пользователя
    size_t count = 10000;
    std::cout << "Введите размер массива целых чисел (по умолчанию 10000): ";
    if (!(std::cin >> count) || count == 0) {
        count = 10000;
    }

    size_t sizeInBytes = count * sizeof(int);
    std::cout << "Запрошено элементов: " << count 
              << " (" << sizeInBytes << " байт / ~" << (sizeInBytes / 1024) << " КБ)" << std::endl;

    // 2. Выделение страниц виртуальной памяти напрямую у ядра Windows через VirtualAlloc
    int* pArray = static_cast<int*>(VirtualAlloc(
        NULL,                   // Ядро само выбирает базовый адрес в адресном пространстве
        sizeInBytes,            // Объем выделяемой памяти в байтах
        MEM_COMMIT | MEM_RESERVE, // Резервирование адресного диапазона и выделение физических страниц
        PAGE_READWRITE          // Права доступа: чтение и запись
    ));

    if (pArray == NULL) {
        std::cout << "Ошибка выделения памяти у ядра: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Память успешно выделена ядром по адресу: " << static_cast<void*>(pArray) << std::endl;

    // 3. Заполнение массива квадратами индексов: arr[i] = i * i
    for (size_t i = 0; i < count; ++i) {
        pArray[i] = static_cast<int>(i * i);
    }

    // 4. Вывод первых 10 элементов массива на экран
    std::cout << "\nПервые 10 элементов массива (квадраты индексов):" << std::endl;
    for (size_t i = 0; i < 10 && i < count; ++i) {
        std::cout << "  pArray[" << i << "] = " << pArray[i] << std::endl;
    }

    // 5. Освобождение памяти напрямую ядру ОС через VirtualFree
    BOOL freeSuccess = VirtualFree(
        pArray,         // Базовый адрес освобождаемого блока
        0,              // Для MEM_RELEASE размер обязан быть равен 0
        MEM_RELEASE     // Полное освобождение физических страниц и де-резервирование виртуального диапазона
    );

    if (freeSuccess) {
        std::cout << "\nПамять успешно возвращена ядру Windows (VirtualFree MEM_RELEASE)." << std::endl;
    } else {
        std::cout << "\nОшибка освобождения памяти: " << GetLastError() << std::endl;
    }

    return 0;
}

Без проверки дальше было бы разыменование NULL.

Как запустить — задача №3
Файлы не нужны вообще. Программа не открывает и не создаёт файлов — только память и консоль. Единственное условие запуска — консоль (cmd или PowerShell), потому что программа ждёт ввода числа с клавиатуры.

Как вводить размер: после приглашения введите число и нажмите Enter. Примеры: 10000 (по умолчанию), 1000, 500000. Чтобы получить значение по умолчанию, введите 0 или любую букву — просто нажать Enter не выйдет: оператор >> пропускает пустые строки и ждёт число.

Ожидаемый вывод (для 10000):

Введите размер массива целых чисел (по умолчанию 10000): 10000
Запрошено элементов: 10000 (40000 байт / ~39 КБ)
Память успешно выделена ядром по адресу: 0x000001...

Первые 10 элементов массива (квадраты индексов):
  pArray[0] = 0
  pArray[1] = 1
  pArray[2] = 4
  pArray[3] = 9
  pArray[4] = 16
  pArray[5] = 25
  pArray[6] = 36
  pArray[7] = 49
  pArray[8] = 64
  pArray[9] = 81

Память успешно возвращена ядру Windows (VirtualFree MEM_RELEASE).

Ограничения ввода: при count больше 46 341 значения i*i перестают влезать в int (переполнение) — для учебного запуска берите count ≤ 46 341. Гигантские значения (например, больше свободной оперативной памяти) дадут «Ошибка выделения памяти у ядра» и завершение с кодом 1 — это честная проверка из программы, а не падение. Реально выделяемый объём всегда округляется ядром вверх до целого числа страниц по 4096 байт.

Общие условия запуска — все три программы
Windows 10/11 x64
Общее для всех трёх программ. Это программы только для Windows (заголовок <windows.h>, вызовы Win32 API, команда chcp) — на Linux или macOS они не скомпилируются. Компиляция — из папки lab5c++, запуск — из командной строки, чтобы видеть вывод и вводить данные.
REM MSVC (Developer Command Prompt for VS):
cl /EHsc task1_no_buffering_bench.cpp
cl /EHsc task2_dbms_sim_filemapping_tlb.cpp
cl /EHsc task3_direct_kernel_memory.cpp

REM MinGW-w64 (PowerShell или cmd):
g++ -O2 -o task1.exe task1_no_buffering_bench.cpp
g++ -O2 -o task2.exe task2_dbms_sim_filemapping_tlb.cpp
g++ -O2 -o task3.exe task3_direct_kernel_memory.cpp

Флаг /EHsc у MSVC включает обработку исключений C++ (нужен для <iostream> и <vector>). Отдельные библиотеки подключать не надо: все вызовы (CreateFileW, VirtualAlloc, MapViewOfFile) живут в kernel32.lib, которая линкуется по умолчанию. Запуск — из той же папки, где лежит exe:

REM cmd:
task1_no_buffering_bench.exe
REM PowerShell:
.\task1_no_buffering_bench.exe

Запуск двойным кликом по exe тоже работает, но окно консоли закроется сразу после завершения — вывод не прочитать. Для задачи 3 консоль обязательна: программа ждёт ввода числа с клавиатуры.

Рабочая папка. Файлы test_direct_io_10mb.dat и database_16mb.bin создаются в текущей рабочей папке — там, откуда запущен exe. Если запускать exe из другой папки (например, через ярлык с другим «Рабочая папка»), файлы появятся там, а не рядом с exe. Проверяйте: после запуска рядом с exe должны появиться .dat (10 МБ) и .bin (16 МБ).
