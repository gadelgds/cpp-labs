#include <windows.h>
#include <iostream>
#include <vector>

const DWORD FILE_SIZE_10MB = 10 * 1024 * 1024;
const DWORD SECTOR_SIZE = 4096;

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    const wchar_t* testFile = L"test_direct_io_10mb.dat";

    // 1. Демонстрация ошибки 87 на невыровненном буфере
    HANDLE hDirect = CreateFileW(testFile, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, 
                                 FILE_FLAG_NO_INTERMEDIATE_BUFFERING, NULL);

    char unalignedBuf[1000]; // Не выровнен по границе 4096 байт!
    DWORD bytesRead = 0;
    BOOL result = ReadFile(hDirect, unalignedBuf, sizeof(unalignedBuf), &bytesRead, NULL);
    if (!result && GetLastError() == ERROR_INVALID_PARAMETER) {
        std::wcout << L"Получена ошибка 87 (ERROR_INVALID_PARAMETER): буфер не выровнен!" << std::endl;
    }
    CloseHandle(hDirect);

    // 2. Исправление через VirtualAlloc (выравнивание кратно 4096)
    const DWORD alignedChunkSize = 64 * 1024; // 64 КБ
    void* alignedBuffer = VirtualAlloc(NULL, alignedChunkSize, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);

    // Успешное чтение выровненным буфером...
    VirtualFree(alignedBuffer, 0, MEM_RELEASE);
    return 0;
}
Подробный построчный разбор
FILE_FLAG_NO_INTERMEDIATE_BUFFERING Прямой доступ к диску в обход системного файлового кэша. Данные передаются напрямую контроллером DMA в буфер приложения.
ERROR_INVALID_PARAMETER (код 87) Возникает, если адрес буфера, файловое смещение или размер чтения не кратны аппаратному размеру сектора диска (4096 байт).
VirtualAlloc(...) Выделяет память напрямую у ядра, гарантируя идеальное выравнивание адреса по границе страницы (4096 байт).
Задача №2: Симуляция СУБД (Случайный доступ и сброс TLB)
task2_dbms_sim_filemapping_tlb.cpp
Причина просадки Режима Б (TLB Flush): При частых вызовах UnmapViewOfFile ядро Windows вынуждено непрерывно перестраивать таблицу страниц (Page Table) процесса и инициировать процессорную инструкцию INVLPG, сбрасывая аппаратный кэш трансляции адресов (TLB Flush). Стандартный I/O с системным кэшем удерживает страницы стабильно в RAM и работает значительно быстрее.
Построчный разбор механизмов

1. Режим А: Вызовы SetFilePointerEx + ReadFile/WriteFile обращаются к системному кэшу страниц Windows без изменений таблиц страниц процесса.

2. Режим Б: При каждом обращении создается проекция MapViewOfFile и немедленно вызывается UnmapViewOfFile. Это инвалидирует записи в TLB процессора, вызывая промахи (TLB Miss) и системные прерывания Page Fault при следующем обращении.

Задача №3: Выделение памяти напрямую у ядра Windows (VirtualAlloc)
task3_direct_kernel_memory.cpp
task3_direct_kernel_memory.cpp 
#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    size_t count = 10000;
    size_t sizeInBytes = count * sizeof(int);

    // 1. Выделение страниц виртуальной памяти напрямую у ядра Windows
    int* pArray = static_cast<int*>(VirtualAlloc(
        NULL,
        sizeInBytes,
        MEM_COMMIT | MEM_RESERVE,
        PAGE_READWRITE
    ));

    // 2. Заполнение квадратами индексов: arr[i] = i * i
    for (size_t i = 0; i < count; ++i) {
        pArray[i] = static_cast<int>(i * i);
    }

    // 3. Вывод первых 10 элементов
    for (size_t i = 0; i < 10; ++i) {
        std::cout << "pArray[" << i << "] = " << pArray[i] << std::endl;
    }

    // 4. Освобождение памяти ядру (для MEM_RELEASE размер обязан быть равен 0!)
    VirtualFree(pArray, 0, MEM_RELEASE);

    return 0;
}
Построчный разбор параметров вызова
MEM_RESERVE Резервирует непрерывный диапазон адресов в 64-битном виртуальном пространстве процесса без выделения физической RAM.
MEM_COMMIT Привязывает физические страницы оперативной памяти (или файл подкачки) к зарезервированному адресному диапазону.
VirtualFree(pArray, 0, MEM_RELEASE); Полный возврат страниц ядру Windows. При флаге MEM_RELEASE параметр размера обязан быть строго равен 0.
