#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <vector>
#include <random>

const DWORD FILE_SIZE_10MB = 10 * 1024 * 1024; // 10 МБ
const DWORD SECTOR_SIZE = 4096;                // 4 КБ (размер сектора диска)

// Создание тестового файла на 10 МБ со случайными символами
bool GenerateRandomFile(const wchar_t* path, DWORD sizeBytes) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    const DWORD chunkSize = 64 * 1024;
    std::vector<char> buffer(chunkSize);
    std::mt19937 rng(42);
    for (size_t i = 0; i < chunkSize; ++i) buffer[i] = static_cast<char>('A' + (rng() % 26));

    DWORD writtenTotal = 0;
    while (writtenTotal < sizeBytes) {
        DWORD toWrite = min(chunkSize, sizeBytes - writtenTotal);
        DWORD written = 0;
        WriteFile(hFile, buffer.data(), toWrite, &written, NULL);
        writtenTotal += written;
    }
    CloseHandle(hFile);
    return true;
}

// Замер времени чтения через QueryPerformanceCounter
double BenchmarkReading(const wchar_t* path, DWORD flags, void* buffer, DWORD readChunkSize) {
    HANDLE hFile = CreateFileW(path, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, flags, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return -1.0;

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&start);

    DWORD bytesRead = 0;
    DWORD totalRead = 0;
    while (totalRead < FILE_SIZE_10MB) {
        if (!ReadFile(hFile, buffer, readChunkSize, &bytesRead, NULL) || bytesRead == 0) {
            break;
        }
        totalRead += bytesRead;
    }

    QueryPerformanceCounter(&end);
    CloseHandle(hFile);

    return static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
}

    }

    // 3. Исправление ошибки: выделение памяти через VirtualAlloc (выравнивание по 4096 байт)
    std::wcout << L"\n3. Выделение выровненного буфера через VirtualAlloc (64 КБ)..." << std::endl;
    const DWORD alignedChunkSize = 64 * 1024; // 64 КБ (кратно 4096)
    void* alignedBuffer = VirtualAlloc(NULL, alignedChunkSize, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);

    if (!alignedBuffer) {
        std::wcout << L"Ошибка VirtualAlloc!" << std::endl;
        return 1;
    }
    std::wcout << L"   Адрес выровненного буфера: " << alignedBuffer 
               << L" (Адрес % 4096 == " << ((UINT_PTR)alignedBuffer % SECTOR_SIZE) << L")" << std::endl;

    // 4. Сравнение скорости чтения с кэшированием и без кэширования
    std::wcout << L"\n4. Сравнение производительности чтения 10 МБ (QueryPerformanceCounter):" << std::endl;

    // Тест А: Со стандартным кэшированием ОС (FILE_ATTRIBUTE_NORMAL)
    double timeCached = BenchmarkReading(testFile, FILE_ATTRIBUTE_NORMAL, alignedBuffer, alignedChunkSize);
    std::wcout << L"   А) Стандартное чтение (с системным кэшем):  " << timeCached << L" мс" << std::endl;

    // Тест Б: Небуферизированный ввод-вывод (FILE_FLAG_NO_BUFFERING)
    double timeDirect = BenchmarkReading(testFile, FILE_FLAG_NO_BUFFERING, alignedBuffer, alignedChunkSize);
    std::wcout << L"   Б) Прямое чтение (без промежуточного кэша): " << timeDirect << L" мс" << std::endl;

    // Освобождение памяти
    VirtualFree(alignedBuffer, 0, MEM_RELEASE);

    return 0;
}

Как запустить — задача №1
Файл готовить заранее НЕ нужно. Программа сама создаёт test_direct_io_10mb.dat (10 МБ) в рабочей папке при запуске — функция GenerateRandomFile с флагом CREATE_ALWAYS. При повторном запуске старый файл просто перезаписывается. Нужно только: ~10 МБ свободного места на диске и право на запись в папку, откуда запущен exe.

Если удалить .dat-файл вручную — ничего страшного: при следующем запуске он создастся заново. Единственное место, где файл обязан существовать, — внутри BenchmarkReading (OPEN_EXISTING), но к тому моменту он уже создан шагом 1.

Типичные ситуации (не ошибки): код 87 на шаге 2 — это запланированная демонстрация, а не сбой; времена X и Y меняются от запуска к запуску (влияют SSD/HDD, загрузка диска, кэш). Настоящие ошибки: время «-1 мс» — файл не создался (нет прав на запись, диск полон, антивирус заблокировал создание .dat) — проверьте, появился ли test_direct_io_10mb.dat рядом с exe; «Адрес % 4096 == 0» не равен нулю — практически невозможно, VirtualAlloc всегда выравнивает.

Задача №2: Симуляция СУБД (Случайный доступ и сброс TLB)

#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <vector>
#include <random>

const DWORD FILE_SIZE_16MB = 16 * 1024 * 1024; // 16 МБ
const int NUM_OPERATIONS = 10000;              // 10 000 случайных модификаций

// Создание 16 МБ бинарного файла последовательности int
bool CreateDbBinaryFile(const wchar_t* path) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    const size_t intsPerBlock = 64 * 1024;
    std::vector<int> block(intsPerBlock);
    for (size_t i = 0; i < intsPerBlock; ++i) block[i] = static_cast<int>(i);

    DWORD totalInts = FILE_SIZE_16MB / sizeof(int);
    DWORD writtenInts = 0;
    while (writtenInts < totalInts) {
        DWORD toWriteInts = min(intsPerBlock, totalInts - writtenInts);
        DWORD writtenBytes = 0;
        WriteFile(hFile, block.data(), toWriteInts * sizeof(int), &writtenBytes, NULL);
        writtenInts += toWriteInts;
    }
    CloseHandle(hFile);
    return true;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    _setmode(_fileno(stdout), _O_U8TEXT);
    #endif

    const wchar_t* dbFile = L"database_16mb.bin";
    std::wcout << L"=== ПРАКТИКА №2: СИМУЛЯЦИЯ СУБД (СТАНДАРТНЫЙ I/O vs FILE MAPPING & TLB FLUSH) ===" << std::endl;

    std::wcout << L"1. Создание бинарного файла 16 МБ с массивом целых чисел (int)..." << std::endl;
    CreateDbBinaryFile(dbFile);

    // Генерация массива случайных индексов для честного сравнения одинаковых обращений
    std::mt19937 rng(1337);
    DWORD totalIntCount = FILE_SIZE_16MB / sizeof(int);
    std::vector<DWORD> randomIndices(NUM_OPERATIONS);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        randomIndices[i] = rng() % totalIntCount;
    }

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);

    // ==============================================================
    // РЕЖИМ А: СТАНДАРТНЫЙ I/O (SetFilePointerEx + ReadFile / WriteFile)
    // ==============================================================
    std::wcout << L"\n2. Запуск Режима А (Стандартный I/O со системным кэшем): 10 000 операций..." << std::endl;
    HANDLE hFileA = CreateFileW(dbFile, GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ | FILE_SHARE_WRITE,
                                NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);

    QueryPerformanceCounter(&start);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        LARGE_INTEGER offset;
        offset.QuadPart = (LONGLONG)randomIndices[i] * sizeof(int);

        SetFilePointerEx(hFileA, offset, NULL, FILE_BEGIN);
        int value = 0;
        DWORD bytesRead = 0;
        ReadFile(hFileA, &value, sizeof(int), &bytesRead, NULL);

        value += 1; // Модификация записи

        SetFilePointerEx(hFileA, offset, NULL, FILE_BEGIN);
        DWORD bytesWritten = 0;
        WriteFile(hFileA, &value, sizeof(int), &bytesWritten, NULL);
    }
    QueryPerformanceCounter(&end);
    CloseHandle(hFileA);

    double timeModeA = static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
    std::wcout << L"   -> Время выполнения Режима А: " << timeModeA << L" мс" << std::endl;

    // ==============================================================
    // РЕЖИМ Б: СКОЛЬЗЯЩЕЕ ОКНО MapViewOfFile + UnmapViewOfFile (TLB FLUSH)
    // ==============================================================
    std::wcout << L"\n3. Запуск Режима Б (Ручное проецирование с UnmapViewOfFile): 10 000 операций..." << std::endl;
    SYSTEM_INFO si;
    GetSystemInfo(&si);
    DWORD allocGranularity = si.dwAllocationGranularity; // 64 КБ
    DWORD windowSize = allocGranularity;                 // Скользящее окно 64 КБ

    HANDLE hFileB = CreateFileW(dbFile, GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ | FILE_SHARE_WRITE,
                                NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    HANDLE hMappingB = CreateFileMappingW(hFileB, NULL, PAGE_READWRITE, 0, 0, NULL);

    QueryPerformanceCounter(&start);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        LONGLONG byteOffset = (LONGLONG)randomIndices[i] * sizeof(int);

        // Выравнивание смещения на границу 64 КБ
        DWORD alignedOffsetLow = (DWORD)((byteOffset / allocGranularity) * allocGranularity);
        DWORD offsetWithinWindow = (DWORD)(byteOffset % allocGranularity);

        // Проецирование участка файла
        char* pView = static_cast<char*>(MapViewOfFile(hMappingB, FILE_MAP_ALL_ACCESS, 0, alignedOffsetLow, windowSize));
        if (pView) {
            int* pVal = reinterpret_cast<int*>(pView + offsetWithinWindow);
            *pVal += 1; // Модификация в спроецированной памяти

            // Принудительное размонтирование (инвалидация записей в TLB процессора)
            UnmapViewOfFile(pView);
        }
    }
    QueryPerformanceCounter(&end);

    CloseHandle(hMappingB);
    CloseHandle(hFileB);

    double timeModeB = static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
    std::wcout << L"   -> Время выполнения Режима Б: " << timeModeB << L" мс" << std::endl;

    std::wcout << L"\n--- АНАЛИЗ НАКЛАДНЫХ РАСХОДОВ TLB FLUSH ---" << std::endl;
    std::wcout << L"1. При частых вызовах UnmapViewOfFile ядро ОС Windows вынуждено:" << std::endl;
    std::wcout << L"   - Изменять таблицы страниц (Page Tables) процесса;" << std::endl;
    std::wcout << L"   - Выполнять аппаратную инструкцию сброса буфера ассоциативной трансляции (TLB Flush / INVLPG);" << std::endl;
    std::wcout << L"   - Обрабатывать Page Fault при следующем MapViewOfFile." << std::endl;
    std::wcout << L"2. В Режиме А системный файловый кэш держит страницы непрерывно в ОЗУ без переключения маппингов." << std::endl;

    return 0;
}

    std::wcout << L"   -> Время выполнения Режима А: " << timeModeA << L" мс" << std::endl;

    std::wcout << L"\n3. Запуск Режима Б (Ручное проецирование с UnmapViewOfFile): 10 000 операций..." << std::endl;
    SYSTEM_INFO si;
    GetSystemInfo(&si);
    DWORD allocGranularity = si.dwAllocationGranularity;
    DWORD windowSize = allocGranularity;

    HANDLE hFileB = CreateFileW(dbFile, GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ | FILE_SHARE_WRITE,
                                NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    HANDLE hMappingB = CreateFileMappingW(hFileB, NULL, PAGE_READWRITE, 0, 0, NULL);

    QueryPerformanceCounter(&start);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        LONGLONG byteOffset = (LONGLONG)randomIndices[i] * sizeof(int);

        DWORD alignedOffsetLow = (DWORD)((byteOffset / allocGranularity) * allocGranularity);
        DWORD offsetWithinWindow = (DWORD)(byteOffset % allocGranularity);

        char* pView = static_cast<char*>(MapViewOfFile(hMappingB, FILE_MAP_ALL_ACCESS, 0, alignedOffsetLow, windowSize));
        if (pView) {
            int* pVal = reinterpret_cast<int*>(pView + offsetWithinWindow);
            *pVal += 1;

            UnmapViewOfFile(pView);
        }
    }
    QueryPerformanceCounter(&end);

    CloseHandle(hMappingB);
    CloseHandle(hFileB);

    double timeModeB = static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
    std::wcout << L"   -> Время выполнения Режима Б: " << timeModeB << L" мс" << std::endl;

    std::wcout << L"\n--- АНАЛИЗ НАКЛАДНЫХ РАСХОДОВ TLB FLUSH ---" << std::endl;
    std::wcout << L"1. При частых вызовах UnmapViewOfFile ядро ОС Windows вынуждено:" << std::endl;
    std::wcout << L"   - Изменять таблицы страниц (Page Tables) процесса;" << std::endl;
    std::wcout << L"   - Выполнять аппаратную инструкцию сброса буфера ассоциативной трансляции (TLB Flush / INVLPG);" << std::endl;
    std::wcout << L"   - Обрабатывать Page Fault при следующем MapViewOfFile." << std::endl;
    std::wcout << L"2. В Режиме А системный файловый кэш держит страницы непрерывно в ОЗУ без переключения маппингов." << std::endl;

    return 0;
}
Как запустить — задача №2
Файл готовить заранее НЕ нужно. Программа сама создаёт database_16mb.bin (16 МБ, 4 194 304 значения int) в рабочей папке при запуске (CreateDbBinaryFile, CREATE_ALWAYS). Нужно только: ~16 МБ свободного места и право на запись в папку с exe. После прогона файл можно удалить — при следующем запуске он пересоздаётся.

=== ПРАКТИКА №2: СИМУЛЯЦИЯ СУБД (СТАНДАРТНЫЙ I/O vs FILE MAPPING & TLB FLUSH) ===
1. Создание бинарного файла 16 МБ с массивом целых чисел (int)...

2. Запуск Режима А (Стандартный I/O со системным кэшем): 10 000 операций...
   -> Время выполнения Режима А: X мс

3. Запуск Режима Б (Ручное проецирование с UnmapViewOfFile): 10 000 операций...
   -> Время выполнения Режима Б: Y мс

--- АНАЛИЗ НАКЛАДНЫХ РАСХОДОВ TLB FLUSH ---
1. При частых вызовах UnmapViewOfFile ядро ОС Windows вынуждено:
   - Изменять таблицы страниц (Page Tables) процесса;
   - Выполнять аппаратную инструкцию сброса буфера ассоциативной трансляции (TLB Flush / INVLPG);
   - Обрабатывать Page Fault при следующем MapViewOfFile.

Что должно получиться: время режима Б заметно больше времени режима А (обычно в разы) — это и есть измеряемый эффект TLB Flush. Если вдруг Б оказался быстрее — скорее всего, диск/кэш прогреты первым режимом, а второй прогон подряд даст уже другую картину; перезапустите программу. Настоящие ошибки: «Время -1 мс» не предусмотрено в этой программе (время считается всегда), но если файл не создался — оба режима будут читать несуществующий файл и молча работать с ошибками ReadFile/WriteFile; проверьте, что database_16mb.bin появился и его размер — ровно 16 777 216 байт.

Задача №3: Выделение памяти напрямую у ядра Windows (VirtualAlloc)

task3_direct_kernel_memory.cpp
#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ПРАКТИКА №3: ВЫДЕЛЕНИЕ ПАМЯТИ НАПРЯМУЮ У ЯДРА WINDOWS (VirtualAlloc) ===" << std::endl;

    // 1. Запрос размера массива у пользователя
    size_t count = 10000;
    std::cout << "Введите размер массива целых чисел (по умолчанию 10000): ";
    if (!(std::cin >> count) || count == 0) {
        count = 10000;
    }

    size_t sizeInBytes = count * sizeof(int);
    std::cout << "Запрошено элементов: " << count 
              << " (" << sizeInBytes << " байт / ~" << (sizeInBytes / 1024) << " КБ)" << std::endl;

    // 2. Выделение страниц виртуальной памяти напрямую у ядра Windows через VirtualAlloc
    int* pArray = static_cast<int*>(VirtualAlloc(
        NULL,                   // Ядро само выбирает базовый адрес в адресном пространстве
        sizeInBytes,            // Объем выделяемой памяти в байтах
        MEM_COMMIT | MEM_RESERVE, // Резервирование адресного диапазона и выделение физических страниц
        PAGE_READWRITE          // Права доступа: чтение и запись
    ));

    if (pArray == NULL) {
        std::cout << "Ошибка выделения памяти у ядра: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Память успешно выделена ядром по адресу: " << static_cast<void*>(pArray) << std::endl;

    // 3. Заполнение массива квадратами индексов: arr[i] = i * i
    for (size_t i = 0; i < count; ++i) {
        pArray[i] = static_cast<int>(i * i);
    }

    // 4. Вывод первых 10 элементов массива на экран
    std::cout << "\nПервые 10 элементов массива (квадраты индексов):" << std::endl;
    for (size_t i = 0; i < 10 && i < count; ++i) {
        std::cout << "  pArray[" << i << "] = " << pArray[i] << std::endl;
    }

    // 5. Освобождение памяти напрямую ядру ОС через VirtualFree
    BOOL freeSuccess = VirtualFree(
        pArray,         // Базовый адрес освобождаемого блока
        0,              // Для MEM_RELEASE размер обязан быть равен 0
        MEM_RELEASE     // Полное освобождение физических страниц и де-резервирование виртуального диапазона
    );

    if (freeSuccess) {
        std::cout << "\nПамять успешно возвращена ядру Windows (VirtualFree MEM_RELEASE)." << std::endl;
    } else {
        std::cout << "\nОшибка освобождения памяти: " << GetLastError() << std::endl;
    }

    return 0;
}

Запуск двойным кликом по exe тоже работает, но окно консоли закроется сразу после завершения — вывод не прочитать. Для задачи 3 консоль обязательна: программа ждёт ввода числа с клавиатуры.

Проверяйте: после запуска рядом с exe должны появиться .dat (10 МБ) и .bin (16 МБ).
