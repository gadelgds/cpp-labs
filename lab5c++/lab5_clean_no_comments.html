#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <vector>
#include <random>

const DWORD FILE_SIZE_10MB = 10 * 1024 * 1024;
const DWORD SECTOR_SIZE = 4096;

bool GenerateRandomFile(const wchar_t* path, DWORD sizeBytes) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    const DWORD chunkSize = 64 * 1024;
    std::vector<char> buffer(chunkSize);
    std::mt19937 rng(42);
    for (size_t i = 0; i < chunkSize; ++i) buffer[i] = static_cast<char>('A' + (rng() % 26));

    DWORD writtenTotal = 0;
    while (writtenTotal < sizeBytes) {
        DWORD toWrite = min(chunkSize, sizeBytes - writtenTotal);
        DWORD written = 0;
        WriteFile(hFile, buffer.data(), toWrite, &written, NULL);
        writtenTotal += written;
    }
    CloseHandle(hFile);
    return true;
}

double BenchmarkReading(const wchar_t* path, DWORD flags, void* buffer, DWORD readChunkSize) {
    HANDLE hFile = CreateFileW(path, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING, flags, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return -1.0;

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&start);

    DWORD bytesRead = 0;
    DWORD totalRead = 0;
    while (totalRead < FILE_SIZE_10MB) {
        if (!ReadFile(hFile, buffer, readChunkSize, &bytesRead, NULL) || bytesRead == 0) {
            break;
        }
        totalRead += bytesRead;
    }

    QueryPerformanceCounter(&end);
    CloseHandle(hFile);

    return static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    _setmode(_fileno(stdout), _O_U8TEXT);
    #endif

    const wchar_t* testFile = L"test_direct_io_10mb.dat";
    std::wcout << L"=== ПРАКТИКА №1: НЕБУФЕРИЗИРОВАННЫЙ ВВОД-ВЫВОД (NO_BUFFERING) ===" << std::endl;

    std::wcout << L"1. Создание тестового файла 10 МБ со случайными данными..." << std::endl;
    GenerateRandomFile(testFile, FILE_SIZE_10MB);

    std::wcout << L"\n2. Попытка чтения в невыровненный буфер (char unalignedBuf[1000])..." << std::endl;
    HANDLE hDirect = CreateFileW(testFile, GENERIC_READ, FILE_SHARE_READ, NULL, OPEN_EXISTING,
                                 FILE_FLAG_NO_BUFFERING, NULL);

    if (hDirect != INVALID_HANDLE_VALUE) {
        char unalignedBuf[1000];
        DWORD bytesRead = 0;
        BOOL result = ReadFile(hDirect, unalignedBuf, sizeof(unalignedBuf), &bytesRead, NULL);
        if (!result) {
            DWORD err = GetLastError();
            std::wcout << L"   [ОШИБКА ПОЛУЧЕНА]: ReadFile вернул FALSE, GetLastError() = " << err << std::endl;
            if (err == ERROR_INVALID_PARAMETER) {
                std::wcout << L"   -> Код 87 (ERROR_INVALID_PARAMETER): буфер не выровнен по границе сектора (4096 байт)!" << std::endl;
            }
        }
        CloseHandle(hDirect);
    }

    std::wcout << L"\n3. Выделение выровненного буфера через VirtualAlloc (64 КБ)..." << std::endl;
    const DWORD alignedChunkSize = 64 * 1024;
    void* alignedBuffer = VirtualAlloc(NULL, alignedChunkSize, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);

    if (!alignedBuffer) {
        std::wcout << L"Ошибка VirtualAlloc!" << std::endl;
        return 1;
    }
    std::wcout << L"   Адрес выровненного буфера: " << alignedBuffer
               << L" (Адрес % 4096 == " << ((UINT_PTR)alignedBuffer % SECTOR_SIZE) << L")" << std::endl;

    std::wcout << L"\n4. Сравнение производительности чтения 10 МБ (QueryPerformanceCounter):" << std::endl;

    double timeCached = BenchmarkReading(testFile, FILE_ATTRIBUTE_NORMAL, alignedBuffer, alignedChunkSize);
    std::wcout << L"   А) Стандартное чтение (с системным кэшем):  " << timeCached << L" мс" << std::endl;

    double timeDirect = BenchmarkReading(testFile, FILE_FLAG_NO_BUFFERING, alignedBuffer, alignedChunkSize);
    std::wcout << L"   Б) Прямое чтение (без промежуточного кэша): " << timeDirect << L" мс" << std::endl;

    VirtualFree(alignedBuffer, 0, MEM_RELEASE);

    return 0;
}

Задача 2. Симуляция СУБД: обычный I/O против file mapping и TLB flush

Чистое решение (без комментариев)

#include <windows.h>
#include <iostream>
#include <io.h>
#include <fcntl.h>
#include <vector>
#include <random>

const DWORD FILE_SIZE_16MB = 16 * 1024 * 1024;
const int NUM_OPERATIONS = 10000;

bool CreateDbBinaryFile(const wchar_t* path) {
    HANDLE hFile = CreateFileW(path, GENERIC_WRITE, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);
    if (hFile == INVALID_HANDLE_VALUE) return false;

    const size_t intsPerBlock = 64 * 1024;
    std::vector<int> block(intsPerBlock);
    for (size_t i = 0; i < intsPerBlock; ++i) block[i] = static_cast<int>(i);

    DWORD totalInts = FILE_SIZE_16MB / sizeof(int);
    DWORD writtenInts = 0;
    while (writtenInts < totalInts) {
        DWORD toWriteInts = min(intsPerBlock, totalInts - writtenInts);
        DWORD writtenBytes = 0;
        WriteFile(hFile, block.data(), toWriteInts * sizeof(int), &writtenBytes, NULL);
        writtenInts += toWriteInts;
    }
    CloseHandle(hFile);
    return true;
}

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    _setmode(_fileno(stdout), _O_U8TEXT);
    #endif

    const wchar_t* dbFile = L"database_16mb.bin";
    std::wcout << L"=== ПРАКТИКА №2: СИМУЛЯЦИЯ СУБД (СТАНДАРТНЫЙ I/O vs FILE MAPPING & TLB FLUSH) ===" << std::endl;

    std::wcout << L"1. Создание бинарного файла 16 МБ с массивом целых чисел (int)..." << std::endl;
    CreateDbBinaryFile(dbFile);

    std::mt19937 rng(1337);
    DWORD totalIntCount = FILE_SIZE_16MB / sizeof(int);
    std::vector<DWORD> randomIndices(NUM_OPERATIONS);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        randomIndices[i] = rng() % totalIntCount;
    }

    LARGE_INTEGER freq, start, end;
    QueryPerformanceFrequency(&freq);

    std::wcout << L"\n2. Запуск Режима А (Стандартный I/O со системным кэшем): 10 000 операций..." << std::endl;
    HANDLE hFileA = CreateFileW(dbFile, GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ | FILE_SHARE_WRITE,
                                NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);

    QueryPerformanceCounter(&start);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        LARGE_INTEGER offset;
        offset.QuadPart = (LONGLONG)randomIndices[i] * sizeof(int);

        SetFilePointerEx(hFileA, offset, NULL, FILE_BEGIN);
        int value = 0;
        DWORD bytesRead = 0;
        ReadFile(hFileA, &value, sizeof(int), &bytesRead, NULL);

        value += 1;

        SetFilePointerEx(hFileA, offset, NULL, FILE_BEGIN);
        DWORD bytesWritten = 0;
        WriteFile(hFileA, &value, sizeof(int), &bytesWritten, NULL);
    }
    QueryPerformanceCounter(&end);
    CloseHandle(hFileA);

    double timeModeA = static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
    std::wcout << L"   -> Время выполнения Режима А: " << timeModeA << L" мс" << std::endl;

    std::wcout << L"\n3. Запуск Режима Б (Ручное проецирование с UnmapViewOfFile): 10 000 операций..." << std::endl;
    SYSTEM_INFO si;
    GetSystemInfo(&si);
    DWORD allocGranularity = si.dwAllocationGranularity;
    DWORD windowSize = allocGranularity;

    HANDLE hFileB = CreateFileW(dbFile, GENERIC_READ | GENERIC_WRITE, FILE_SHARE_READ | FILE_SHARE_WRITE,
                                NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    HANDLE hMappingB = CreateFileMappingW(hFileB, NULL, PAGE_READWRITE, 0, 0, NULL);

    QueryPerformanceCounter(&start);
    for (int i = 0; i < NUM_OPERATIONS; ++i) {
        LONGLONG byteOffset = (LONGLONG)randomIndices[i] * sizeof(int);

        DWORD alignedOffsetLow = (DWORD)((byteOffset / allocGranularity) * allocGranularity);
        DWORD offsetWithinWindow = (DWORD)(byteOffset % allocGranularity);

        char* pView = static_cast<char*>(MapViewOfFile(hMappingB, FILE_MAP_ALL_ACCESS, 0, alignedOffsetLow, windowSize));
        if (pView) {
            int* pVal = reinterpret_cast<int*>(pView + offsetWithinWindow);
            *pVal += 1;

            UnmapViewOfFile(pView);
        }
    }
    QueryPerformanceCounter(&end);

    CloseHandle(hMappingB);
    CloseHandle(hFileB);

    double timeModeB = static_cast<double>(end.QuadPart - start.QuadPart) * 1000.0 / freq.QuadPart;
    std::wcout << L"   -> Время выполнения Режима Б: " << timeModeB << L" мс" << std::endl;

    std::wcout << L"\n--- АНАЛИЗ НАКЛАДНЫХ РАСХОДОВ TLB FLUSH ---" << std::endl;
    std::wcout << L"1. При частых вызовах UnmapViewOfFile ядро ОС Windows вынуждено:" << std::endl;
    std::wcout << L"   - Изменять таблицы страниц (Page Tables) процесса;" << std::endl;
    std::wcout << L"   - Выполнять аппаратную инструкцию сброса буфера ассоциативной трансляции (TLB Flush / INVLPG);" << std::endl;
    std::wcout << L"   - Обрабатывать Page Fault при следующем MapViewOfFile." << std::endl;
    std::wcout << L"2. В Режиме А системный файловый кэш держит страницы непрерывно в ОЗУ без переключения маппингов." << std::endl;

    return 0;
}

Задача 3. Выделение памяти напрямую у ядра Windows

Чистое решение (без комментариев)
task3_direct_kernel_memory.cpp
task3_direct_kernel_memory.cpp 
#include <windows.h>
#include <iostream>

int main() {
    #if defined(_WIN32)
    system("chcp 65001 > nul");
    #endif

    std::cout << "=== ПРАКТИКА №3: ВЫДЕЛЕНИЕ ПАМЯТИ НАПРЯМУЮ У ЯДРА WINDOWS (VirtualAlloc) ===" << std::endl;

    size_t count = 10000;
    std::cout << "Введите размер массива целых чисел (по умолчанию 10000): ";
    if (!(std::cin >> count) || count == 0) {
        count = 10000;
    }

    size_t sizeInBytes = count * sizeof(int);
    std::cout << "Запрошено элементов: " << count
              << " (" << sizeInBytes << " байт / ~" << (sizeInBytes / 1024) << " КБ)" << std::endl;

    int* pArray = static_cast<int*>(VirtualAlloc(
        NULL,
        sizeInBytes,
        MEM_COMMIT | MEM_RESERVE,
        PAGE_READWRITE
    ));

    if (pArray == NULL) {
        std::cout << "Ошибка выделения памяти у ядра: " << GetLastError() << std::endl;
        return 1;
    }

    std::cout << "Память успешно выделена ядром по адресу: " << static_cast<void*>(pArray) << std::endl;

    for (size_t i = 0; i < count; ++i) {
        pArray[i] = static_cast<int>(i * i);
    }

    std::cout << "\nПервые 10 элементов массива (квадраты индексов):" << std::endl;
    for (size_t i = 0; i < 10 && i < count; ++i) {
        std::cout << "  pArray[" << i << "] = " << pArray[i] << std::endl;
    }

    BOOL freeSuccess = VirtualFree(
        pArray,
        0,
        MEM_RELEASE
    );

    if (freeSuccess) {
        std::cout << "\nПамять успешно возвращена ядру Windows (VirtualFree MEM_RELEASE)." << std::endl;
    } else {
        std::cout << "\nОшибка освобождения памяти: " << GetLastError() << std::endl;
    }

    return 0;
}

Чем VirtualAlloc отличается от new/malloc? new и malloc берут память из кучи процесса (менеджер кучи сам вызывает VirtualAlloc большими блоками и нарезает их). Прямой VirtualAlloc даёт:
• выравнивание по границе 64 КБ;
• страничную защиту (PAGE_READWRITE, PAGE_NOACCESS и т. д.);
• возможность резервировать и коммитить память по отдельности;
• гарантию, что память вернётся ОС, а не останется в куче.
